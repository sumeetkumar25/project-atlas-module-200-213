"""
Module 203 - Feature Predictive Power Report
"""

from datetime import datetime
import pandas as pd


class FeaturePredictivePowerReport:
    """Report generator for feature predictive power results."""
    
    @staticmethod
    def generate(statistics, metadata=None, dataset_name="Unknown") -> str:
        """Generate a formatted report."""
        lines = []
        
        lines.append("=" * 55)
        lines.append("MODULE 203")
        lines.append("FEATURE PREDICTIVE POWER")
        lines.append("=" * 55)
        lines.append(f"\nDataset: {dataset_name}")
        
        # Overview
        lines.append("\n" + "-" * 45)
        lines.append("OVERVIEW")
        lines.append("-" * 45)
        lines.append(f"  Total Features: {statistics.total_features}")
        lines.append(f"  KEPT: {statistics.kept_count}")
        lines.append(f"  REMOVED: {statistics.removed_count}")
        
        # Percentages
        lines.append("\n" + "-" * 45)
        lines.append("PERCENTAGES")
        lines.append("-" * 45)
        lines.append(f"  KEPT: {statistics.kept_percentage:.1f}%")
        lines.append(f"  REMOVED: {statistics.removed_percentage:.1f}%")
        
        # Predictive Scores
        lines.append("\n" + "-" * 45)
        lines.append("PREDICTIVE SCORES")
        lines.append("-" * 45)
        lines.append(f"  Average: {statistics.average_predictive_score:.3f}")
        lines.append(f"  Median: {statistics.median_predictive_score:.3f}")
        lines.append(f"  Best: {statistics.max_predictive_score:.3f}")
        lines.append(f"  Worst: {statistics.min_predictive_score:.3f}")
        
        # Best and Worst Features
        if statistics.best_feature:
            lines.append("\n" + "-" * 45)
            lines.append("BEST FEATURE")
            lines.append("-" * 45)
            lines.append(f"  {statistics.best_feature}: {statistics.max_predictive_score:.3f}")
        
        if statistics.worst_feature:
            lines.append("\n" + "-" * 45)
            lines.append("WORST FEATURE")
            lines.append("-" * 45)
            lines.append(f"  {statistics.worst_feature}: {statistics.min_predictive_score:.3f}")
        
        # Predictive Metrics
        lines.append("\n" + "-" * 45)
        lines.append("PREDICTIVE METRICS")
        lines.append("-" * 45)
        lines.append(f"  Average IC: {statistics.average_ic:.4f}")
        lines.append(f"  Average Rank IC: {statistics.average_rank_ic:.4f}")
        lines.append(f"  Average Mutual Information: {statistics.average_mi:.4f}")
        
        # Interpretation
        lines.append("\n" + "-" * 45)
        lines.append("INTERPRETATION")
        lines.append("-" * 45)
        
        if statistics.average_predictive_score > 0.5:
            lines.append("  ✅ Features show strong predictive power")
        elif statistics.average_predictive_score > 0.3:
            lines.append("  ✓ Features show moderate predictive power")
        else:
            lines.append("  ⚠ Features show weak predictive power")
        
        if statistics.average_ic > 0.05:
            lines.append("  ✅ Features have positive average IC")
        else:
            lines.append("  ⚠ Features have low average IC")
        
        # Validation
        lines.append("\n" + "-" * 45)
        lines.append("VALIDATION")
        lines.append("-" * 45)
        lines.append("  Status: PASSED")
        
        # Footer
        lines.append(f"\n  Generated: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
        lines.append("\n" + "=" * 55)
        lines.append("END OF REPORT")
        lines.append("=" * 55)
        
        return "\n".join(lines)
    
    @staticmethod
    def generate_score_table(feature_scores: pd.DataFrame, top_n: int = 10) -> str:
        """Generate a score table report."""
        lines = []
        lines.append("\n" + "=" * 80)
        lines.append(f"TOP {top_n} PREDICTIVE FEATURES")
        lines.append("=" * 80)
        
        top_features = feature_scores.head(top_n)
        
        lines.append(
            f"\n{'Feature':<20} {'Score':<12} {'Decision':<12} {'IC':<12} {'Rank IC':<12} {'MI':<12}"
        )
        lines.append("-" * 80)
        
        for _, row in top_features.iterrows():
            lines.append(
                f"{row['feature'][:19]:<20} "
                f"{row['predictive_score']:<12.3f} "
                f"{row['decision']:<12} "
                f"{row['ic']:<12.4f} "
                f"{row['rank_ic']:<12.4f} "
                f"{row['mutual_information']:<12.4f}"
            )
        
        lines.append("\n" + "=" * 80)
        return "\n".join(lines)
    
    @staticmethod
    def save(report: str, filepath: str):
        """Save report to file."""
        from pathlib import Path
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        with open(path, "w", encoding="utf-8") as f:
            f.write(report)
