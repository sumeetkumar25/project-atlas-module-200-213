def create_test_matrix(n=500, n_features=10):
    """Create a synthetic feature matrix for testing."""
    np.random.seed(42)
    
    timestamps = pd.date_range('2020-01-01', periods=n, freq='1h')
    
    df = pd.DataFrame({'timestamp': timestamps})
    
    # Add close price for future returns (as pandas Series)
    close = pd.Series(
        100 + np.cumsum(np.random.randn(n) * 0.5),
        index=df.index
    )
    df['close'] = close
    
    # Predictive features (correlated with future returns)
    df['predictive_feature'] = (
        close.shift(-1) - close +
        np.random.randn(n) * 0.5
    )
    
    # Weakly predictive feature
    df['weak_feature'] = (
        np.random.randn(n) * 0.5 +
        close.shift(-1) - close
    )
    
    # Non-predictive feature
    df['noise_feature'] = np.random.randn(n) * 10
    
    # More features
    for i in range(4, n_features + 1):
        df[f'feature_{i}'] = np.random.randn(n) * 10 + 50
    
    return df


def test_engine_load():
    """Test 1: Load stable matrix."""
    print("\n" + "=" * 60)
    print("TEST 1: Load Stable Matrix")
    print("=" * 60)
    
    df = create_test_matrix()
    engine = FeaturePredictivePowerEngine()
    
    with tempfile.TemporaryDirectory() as tmpdir:
        tmp_path = Path(tmpdir)
        matrix_path = tmp_path / "stable_feature_matrix.parquet"
        df.to_parquet(matrix_path, index=False)
        
        loaded_df = engine.load_stable_matrix(matrix_path)
        
        assert not loaded_df.empty
        assert 'timestamp' in loaded_df.columns
        assert len(loaded_df) == len(df)
    
    print("  ✓ Load test passed")
    return True


def test_engine_future_returns():
    """Test 2: Future returns computation."""
    print("\n" + "=" * 60)
    print("TEST 2: Future Returns")
    print("=" * 60)
    
    df = create_test_matrix()
    engine = FeaturePredictivePowerEngine()
    
    future_returns = engine.compute_future_returns(df, horizon=1)
    
    assert len(future_returns) == len(df)
    assert np.isnan(future_returns[-1])  # Last value should be NaN
    
    print("  ✓ Future returns test passed")
    return True


def test_engine_information_coefficient():
    """Test 3: Information Coefficient."""
    print("\n" + "=" * 60)
    print("TEST 3: Information Coefficient")
    print("=" * 60)
    
    df = create_test_matrix()
    engine = FeaturePredictivePowerEngine()
    
    future_returns = engine.compute_future_returns(df, horizon=1)
    
    # Predictive feature should have positive IC
    ic_result = engine.compute_information_coefficient(df['predictive_feature'], future_returns)
    assert ic_result['ic'] > 0.01
    
    # Noise feature should have near-zero IC
    ic_result = engine.compute_information_coefficient(df['noise_feature'], future_returns)
    assert abs(ic_result['ic']) < 0.1
    
    print("  ✓ IC test passed")
    return True
