# atlas/modules/module203_feature_predictive_power/module.py
"""
Module 203
Feature Predictive Power

Measures how much future market information each stable feature contains.

Pipeline: load stable matrix → compute predictive metrics → score features →
          decisions → save predictive matrix
"""

from typing import Optional, Dict, Any, List
import pandas as pd
from pathlib import Path

from atlas.core.base_research_module import BaseResearchModule
from atlas.framework.registry import AtlasRegistry

from .engine import FeaturePredictivePowerEngine
from .statistics import FeaturePredictivePowerStatisticsEngine
from .validator import run_validation
from .report import FeaturePredictivePowerReport
from .exporter import FeaturePredictivePowerExporter


class FeaturePredictivePowerResearch(BaseResearchModule):
    """Main research module for feature predictive power."""
    
    MODULE_OUTPUT = "module203"
    METADATA = FeaturePredictivePowerStatisticsEngine.METADATA
    
    def __init__(
        self,
        registry: Optional[AtlasRegistry] = None,
        horizons: List[int] = None,
        pass_threshold: float = 0.80,
        warning_threshold: float = 0.60,
        rolling_window: int = 100,
        fdr_alpha: float = 0.05,
        price_column: str = "close",
        timestamp_column: str = "timestamp"
    ):
        """
        Initialize the feature predictive power module.
        
        Args:
            registry: Optional Atlas registry
            horizons: List of forecast horizons
            pass_threshold: Score above this threshold = KEEP
            warning_threshold: Score above this threshold = WARNING
            rolling_window: Window for rolling IC stability
            fdr_alpha: False Discovery Rate significance level
            price_column: Name of the price column
            timestamp_column: Name of the timestamp column
        """
        super().__init__(
            module_name="Feature Predictive Power",
            module_version=self.METADATA["version"],
        )
        
        self.registry = registry or AtlasRegistry()
        self.engine = FeaturePredictivePowerEngine(
            horizons=horizons,
            pass_threshold=pass_threshold,
            warning_threshold=warning_threshold,
            rolling_window=rolling_window,
            fdr_alpha=fdr_alpha,
            timestamp_column=timestamp_column,
            price_column=price_column
        )
        self.stats_engine = FeaturePredictivePowerStatisticsEngine()
        self.result_data = None
        self.statistics = None
        self.score_report = None
        self._output_dir = None
        self._matrix_path = None

    def _register_pipeline_steps(self):
        """Register pipeline steps."""
        self.pipeline_steps = [
            ("feature_predictive_power", self._compute_predictive),
            ("statistics", self._compute_statistics),
            ("validate", self._validate),
            ("export", self._export_results),
        ]
    
    def _prepare_data(self) -> Dict[str, Any]:
        """Prepare data."""
        if self._matrix_path is None:
            raise ValueError("Matrix path not provided. Use run_predictive() method.")
        return {"matrix_path": self._matrix_path}
    
    def _compute_predictive(self) -> Dict[str, Any]:
        """Step 1: Compute feature predictive power."""
        self.logger.info("Step 1: Computing feature predictive power")
        
        prepared = self._prepare_data()
        
        self.result_data = self.engine.compute(
            matrix_path=prepared["matrix_path"],
            output_dir=self._output_dir
        )
        
        self.logger.metric(
            "Total Features",
            str(self.result_data.summary_statistics["total_features"])
        )
        self.logger.metric(
            "KEPT",
            str(self.result_data.summary_statistics["kept_count"])
        )
        self.logger.metric(
            "REMOVED",
            str(self.result_data.summary_statistics["removed_count"])
        )
        self.logger.metric(
            "Avg Predictive Score",
            f"{self.result_data.summary_statistics['average_predictive_score']:.3f}"
        )
        self.logger.metric(
            "Avg IC",
            f"{self.result_data.summary_statistics['average_ic']:.4f}"
        )
        self.logger.success("Feature predictive power computed")
        
        # Generate report
        self.score_report = FeaturePredictivePowerReport.generate_score_table(
            self.result_data.feature_scores,
            top_n=10
        )
        
        return {
            "total_features": self.result_data.summary_statistics["total_features"],
            "kept_count": self.result_data.summary_statistics["kept_count"],
            "removed_count": self.result_data.summary_statistics["removed_count"],
            "average_predictive_score": self.result_data.summary_statistics["average_predictive_score"],
            "average_ic": self.result_data.summary_statistics["average_ic"]
        }
    
    def _compute_statistics(self) -> Dict[str, Any]:
        """Step 2: Computing statistics."""
        self.logger.info("Step 2: Computing statistics")
        self.statistics = self.stats_engine.analyze(self.result_data)
        
        self.logger.metric(
            "Best Feature",
            self.statistics.best_feature
        )
        self.logger.metric(
            "Avg MI",
            f"{self.statistics.average_mi:.4f}"
        )
        self.logger.success("Statistics computed")
        
        return self.stats_engine.summary_dict(self.statistics)
    
    def _validate(self) -> Dict[str, Any]:
        """Step 3: Validation."""
        self.logger.info("Step 3: Validation")
        passed = run_validation(self.result_data, self.statistics)
        self.logger.success("Validation complete")
        return {"software_validation_passed": passed}
    
    def _export_results(self) -> Dict[str, Any]:
        """Step 4: Export."""
        self.logger.info("Step 4: Export")
        
        # Register DataFrames as artifacts
        self.result.artifacts["predictive_matrix"] = self.result_data.predictive_matrix
        self.result.artifacts["feature_scores"] = self.result_data.feature_scores
        self.result.artifacts["feature_rankings"] = self.result_data.feature_rankings
        
        # Store removed features
        removed_df = pd.DataFrame({'removed_features': self.result_data.removed_features})
        self.result.artifacts["removed_features"] = removed_df
        
        # Store summary
        summary_df = pd.DataFrame([self.result_data.summary_statistics])
        self.result.artifacts["prediction_statistics"] = summary_df
        
        # Generate and register reports
        main_report = FeaturePredictivePowerReport.generate(
            self.statistics,
            metadata=self.METADATA,
            dataset_name=self.result.dataset_identifier,
        )
        self.result.artifacts["feature_predictive_power_report"] = main_report
        self.result.artifacts["score_report"] = self.score_report
        
        self.logger.metric("Artifacts", str(len(self.result.artifacts)))
        self.logger.success("Export complete")
        
        return {
            "export_status": "complete",
            "artifact_count": len(self.result.artifacts),
            "kept_count": self.result_data.summary_statistics["kept_count"],
            "removed_count": self.result_data.summary_statistics["removed_count"]
        }
    
    def compute(self) -> Dict[str, Any]:
        """Run the full computation pipeline."""
        self.logger.section("Feature Predictive Power Pipeline")
        results = self.execute_pipeline()
        self.result.status = "completed"
        self.logger.success("Module 203 complete")
        return results
    
    def run(self, filepath: str, output_dir=None):
        """Run module on file."""
        self._output_dir = Path(output_dir) if output_dir else None
        return super().run(filepath, output_dir)
    
    def run_dataframe(self, df, dataset_id="in_memory", output_dir=None):
        """Run module on a single DataFrame."""
        self._output_dir = Path(output_dir) if output_dir else None
        return super().run_dataframe(df, dataset_id, output_dir)
    
    def run_predictive(
        self,
        matrix_path: Path,
        dataset_id: str = "feature_predictive",
        output_dir: Optional[Path] = None
    ) -> Any:
        """
        Run the feature predictive power analysis.
        
        Args:
            matrix_path: Path to the stable feature matrix
            dataset_id: Identifier for the dataset
            output_dir: Output directory for results
        """
        self._matrix_path = matrix_path
        self._output_dir = Path(output_dir) if output_dir else Path("data/feature_store/predictive")
        
        # Create a dummy DataFrame to satisfy BaseResearchModule
        dummy = pd.DataFrame(
            {
                "timestamp": [pd.Timestamp("2000-01-01")]
            }
        )
        
        return super().run_dataframe(
            dummy,
            dataset_id=dataset_id,
            output_dir=self._output_dir
        )
