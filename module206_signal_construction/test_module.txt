def create_test_data(n=100, seed=42):
    """Create synthetic test data for signal construction."""
    np.random.seed(seed)
    
    timestamps = pd.date_range('2020-01-01', periods=n, freq='1h')
    
    # Predictive matrix
    predictive_df = pd.DataFrame({
        'timestamp': timestamps,
        'close': 100 + np.cumsum(np.random.randn(n) * 0.5),
        'feature1': np.random.randn(n) * 10 + 50,
        'feature2': np.random.randn(n) * 5 + 50,
        'feature3': np.random.randn(n) * 8 + 50,
        'feature4': np.random.randn(n) * 3 + 50,
        'feature5': np.random.randn(n) * 6 + 50
    })
    
    # Selected features
    selected_df = pd.DataFrame({
        'selected_features': ['feature1', 'feature2', 'feature3', 'feature4']
    })
    
    # Feature scores
    scores_df = pd.DataFrame({
        'feature': ['feature1', 'feature2', 'feature3', 'feature4', 'feature5'],
        'final_score': [0.9, 0.8, 0.7, 0.6, 0.5]
    })
    
    return {
        'predictive_matrix': predictive_df,
        'selected_features': selected_df,
        'feature_scores': scores_df
    }


def test_engine_load_selected():
    """Test 1: Load selected features."""
    print("\n" + "=" * 60)
    print("TEST 1: Load Selected Features")
    print("=" * 60)
    
    data = create_test_data()
    engine = SignalConstructionEngine()
    
    with tempfile.TemporaryDirectory() as tmpdir:
        tmp_path = Path(tmpdir)
        selected_path = tmp_path / "selected_features.csv"
        data['selected_features'].to_csv(selected_path, index=False)
        
        features = engine.load_selected_features(selected_path)
        
        assert len(features) > 0
        assert 'feature1' in features
    
    print("  ✓ Load selected features test passed")
    return True

def test_engine_normalize_features():
    """Test 5: Feature normalization (no look-ahead bias)."""
    print("\n" + "=" * 60)
    print("TEST 5: Feature Normalization")
    print("=" * 60)
    
    data = create_test_data()
    
    engine = SignalConstructionEngine(
        normalization_method="rolling",
        normalization_window=20
    )
    
    features = ["feature1", "feature2"]
    
    normalized_df = engine.normalize_features(
        data["predictive_matrix"],
        features
    )
    
    for feature in features:
        # After warmup there should be valid values
        valid = normalized_df[feature].dropna()
        
        assert len(valid) > 0
        
        # Variance should exist
        assert valid.std() > 0
        
        # No infinities
        assert np.isfinite(valid).all()
        
        # NaNs are expected because rolling window has not filled yet
        # This verifies no look-ahead bias
        assert normalized_df[feature].isna().sum() > 0
    
    print("  ✓ Feature normalization test passed")
    return True


def test_engine_construct_signal():
    """Test 6: Signal construction."""
    print("\n" + "=" * 60)
    print("TEST 6: Signal Construction")
    print("=" * 60)
    
    data = create_test_data()
    engine = SignalConstructionEngine()
    
    features = ['feature1', 'feature2', 'feature3']
    normalized_df = engine.normalize_features(data['predictive_matrix'], features)
    weights = engine.compute_weights(data['feature_scores'], features)
    
    # Updated to handle tuple return
    signal, available = engine.construct_signal(
        normalized_df,
        weights,
        features
    )
    
    assert len(signal) == len(data['predictive_matrix'])
    assert not signal.isna().any()
    assert len(available) == len(features)
    
    print("  ✓ Signal construction test passed")
    return True
