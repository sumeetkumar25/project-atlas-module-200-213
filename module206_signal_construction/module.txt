"""
Module 206
Signal Construction

Converts selected features into one or more continuous trading signals.

Pipeline: load selected features → load predictive matrix → normalize features →
          compute weights → construct signal → normalize signal → statistics →
          validation → report → export
"""

from typing import Optional, Dict, Any
import pandas as pd
from pathlib import Path

from atlas.core.base_research_module import BaseResearchModule
from atlas.framework.registry import AtlasRegistry

from .engine import SignalConstructionEngine, ScoreWeightingStrategy, RankWeightingStrategy
from .statistics import SignalConstructionStatisticsEngine
from .validator import run_validation
from .report import SignalConstructionReport
from .exporter import SignalConstructionExporter


class SignalConstructionResearch(BaseResearchModule):
    """Main research module for signal construction."""
    
    MODULE_OUTPUT = "module206"
    METADATA = SignalConstructionStatisticsEngine.METADATA
    
    def __init__(
        self,
        registry: Optional[AtlasRegistry] = None,
        weighting_method: str = "score",  # "score" or "rank"
        timestamp_column: str = "timestamp",
        normalization_method: str = "rolling",  # "rolling", "expanding", "global"
        normalization_window: int = 500,
        signal_scaling: str = "tanh",  # "tanh" or "clip"
        handle_missing: bool = True
    ):
        """
        Initialize the signal construction module.
        
        Args:
            registry: Optional Atlas registry
            weighting_method: Method for computing weights ('score' or 'rank')
            timestamp_column: Name of the timestamp column
            normalization_method: Method for feature normalization ('rolling', 'expanding', 'global')
            normalization_window: Window size for rolling normalization
            signal_scaling: Method for scaling signal ('tanh' or 'clip')
            handle_missing: Whether to renormalize weights when features are missing
        """
        super().__init__(
            module_name="Signal Construction",
            module_version=self.METADATA["version"],
        )
        
        self.registry = registry or AtlasRegistry()
        
        # Select weighting strategy
        if weighting_method == "rank":
            weighting_strategy = RankWeightingStrategy()
        else:
            weighting_strategy = ScoreWeightingStrategy()
        
        self.engine = SignalConstructionEngine(
            weighting_strategy=weighting_strategy,
            timestamp_column=timestamp_column,
            normalization_window=normalization_window,
            normalization_method=normalization_method,
            signal_scaling=signal_scaling,
            handle_missing=handle_missing
        )
        self.stats_engine = SignalConstructionStatisticsEngine()
        self.result_data = None
        self.statistics = None
        self.component_report = None
        self._output_dir = None
        self._paths = None

    def _register_pipeline_steps(self):
        """Register pipeline steps."""
        self.pipeline_steps = [
            ("signal_construction", self._construct_signal),
            ("statistics", self._compute_statistics),
            ("validate", self._validate),
            ("export", self._export_results),
        ]
    
    def _prepare_data(self) -> Dict[str, Any]:
        """Prepare data."""
        if self._paths is None:
            raise ValueError("File paths not provided. Use run_signal_construction() method.")
        return self._paths
    
    def _construct_signal(self) -> Dict[str, Any]:
        """Step 1: Construct signal."""
        self.logger.info("Step 1: Constructing signal")
        
        prepared = self._prepare_data()
        
        self.result_data = self.engine.compute(
            selected_features_path=prepared["selected_features"],
            feature_scores_path=prepared["feature_scores"],
            predictive_matrix_path=prepared["predictive_matrix"],
            output_dir=self._output_dir
        )
        
        self.logger.metric(
            "Total Features",
            str(self.result_data.summary_statistics["total_features"])
        )
        self.logger.metric(
            "Signal Mean",
            f"{self.result_data.summary_statistics['signal_mean']:.4f}"
        )
        self.logger.metric(
            "Signal Std",
            f"{self.result_data.summary_statistics['signal_std']:.4f}"
        )
        self.logger.metric(
            "Positive",
            f"{self.result_data.summary_statistics['positive_pct']:.1f}%"
        )
        self.logger.metric(
            "Negative",
            f"{self.result_data.summary_statistics['negative_pct']:.1f}%"
        )
        self.logger.success("Signal constructed")
        
        # Generate component report
        self.component_report = SignalConstructionReport.generate_component_report(
            self.result_data.signal_components,
            top_n=10
        )
        
        return {
            "total_features": self.result_data.summary_statistics["total_features"],
            "signal_mean": self.result_data.summary_statistics["signal_mean"],
            "signal_std": self.result_data.summary_statistics["signal_std"],
            "positive_pct": self.result_data.summary_statistics["positive_pct"],
            "negative_pct": self.result_data.summary_statistics["negative_pct"]
        }
    
    def _compute_statistics(self) -> Dict[str, Any]:
        """Step 2: Computing statistics."""
        self.logger.info("Step 2: Computing statistics")
        self.statistics = self.stats_engine.analyze(self.result_data)
        
        self.logger.metric(
            "Neutral",
            f"{self.statistics.neutral_pct:.1f}%"
        )
        self.logger.metric(
            "Skewness",
            f"{self.statistics.skewness:.3f}"
        )
        self.logger.success("Statistics computed")
        
        return self.stats_engine.summary_dict(self.statistics)
    
    def _validate(self) -> Dict[str, Any]:
        """Step 3: Validation."""
        self.logger.info("Step 3: Validation")
        passed = run_validation(self.result_data, self.statistics)
        self.logger.success("Validation complete")
        return {"software_validation_passed": passed}
    
    def _export_results(self) -> Dict[str, Any]:
        """Step 4: Export."""
        self.logger.info("Step 4: Export")
        
        # Register DataFrames as artifacts
        self.result.artifacts["signal_matrix"] = self.result_data.signal_matrix
        self.result.artifacts["signal_components"] = self.result_data.signal_components
        
        # Store signals as DataFrames
        raw_df = pd.DataFrame({
            'raw_signal': self.result_data.raw_signal.values
        })
        norm_df = pd.DataFrame({
            'normalized_signal': self.result_data.normalized_signal.values
        })
        
        self.result.artifacts["raw_signal"] = raw_df
        self.result.artifacts["normalized_signal"] = norm_df
        
        # Store contribution percentages
        contrib_df = pd.DataFrame({
            'feature': self.result_data.contribution_percentages.index,
            'contribution_percentage': self.result_data.contribution_percentages.values
        }).sort_values('contribution_percentage', ascending=False)
        self.result.artifacts["contribution_percentages"] = contrib_df
        
        # Store summary
        stats_df = pd.DataFrame([self.result_data.summary_statistics])
        self.result.artifacts["signal_statistics"] = stats_df
        
        # Generate and register reports
        main_report = SignalConstructionReport.generate(
            self.statistics,
            metadata=self.METADATA,
            dataset_name=self.result.dataset_identifier,
        )
        self.result.artifacts["signal_construction_report"] = main_report
        self.result.artifacts["component_report"] = self.component_report
        
        self.logger.metric("Artifacts", str(len(self.result.artifacts)))
        self.logger.success("Export complete")
        
        return {
            "export_status": "complete",
            "artifact_count": len(self.result.artifacts),
            "total_features": self.result_data.summary_statistics["total_features"],
            "total_rows": self.result_data.summary_statistics["total_rows"]
        }
    
    def compute(self) -> Dict[str, Any]:
        """Run the full computation pipeline."""
        self.logger.section("Signal Construction Pipeline")
        results = self.execute_pipeline()
        self.result.status = "completed"
        self.logger.success("Module 206 complete")
        return results
    
    def run(self, filepath: str, output_dir=None):
        """Run module on file."""
        self._output_dir = Path(output_dir) if output_dir else None
        return super().run(filepath, output_dir)
    
    def run_dataframe(self, df, dataset_id="in_memory", output_dir=None):
        """Run module on a single DataFrame."""
        self._output_dir = Path(output_dir) if output_dir else None
        return super().run_dataframe(df, dataset_id, output_dir)
    
    def run_signal_construction(
        self,
        selected_features_path: Path,
        feature_scores_path: Path,
        predictive_matrix_path: Path,
        dataset_id: str = "signal_construction",
        output_dir: Optional[Path] = None
    ) -> Any:
        """
        Run the signal construction.
        
        Args:
            selected_features_path: Path to selected_features.csv from Module 205
            feature_scores_path: Path to feature_scores.csv from Module 205
            predictive_matrix_path: Path to predictive_feature_matrix.parquet from Module 203
            dataset_id: Identifier for the dataset
            output_dir: Output directory for results
        """
        self._paths = {
            "selected_features": selected_features_path,
            "feature_scores": feature_scores_path,
            "predictive_matrix": predictive_matrix_path
        }
        self._output_dir = Path(output_dir) if output_dir else Path("data/feature_store/signal")
        
        # Create a dummy DataFrame to satisfy BaseResearchModule
        dummy = pd.DataFrame(
            {
                "timestamp": [pd.Timestamp("2000-01-01")]
            }
        )
        
        return super().run_dataframe(
            dummy,
            dataset_id=dataset_id,
            output_dir=self._output_dir
        )
