"""
Integration Test: Module 108 → Module 109

Tests the complete integration path:
    1. Direct adapter path: 108 → target_adapter → 109
    2. Full dependency path: ResearchRunner → ModuleLoader → 103 → 108
       → ArtifactStore → DependencyInjector → 109

Validates:
    - breakout_* columns are correctly attached as targets
    - breakout_* columns are NOT used as features (no leakage)
    - Module 109 correctly detects features excluding breakout_*
    - Target column 'future_breakout' is properly created
    - Full pipeline executes successfully

FIXED: Module 108 publishes two distinct artifacts with
       different granularity:

         - "features": row-level feature table containing
           timestamp plus the full per-horizon breakout set:
               timestamp
               breakout_1
               breakout_3
               breakout_5
               breakout_10
               breakout_20

         - "state_probabilities": table aggregated by market
           state, and therefore correctly does NOT contain
           breakout_5 (there is no per-row column to expose
           after aggregation).

       Every test reads breakout targets from
       result108.artifacts["features"]. Reading from
       "state_probabilities" would fail the
       `assert "breakout_5" in targets.columns` check on the
       aggregated table. The matching artifact-key assertions
       have also been updated so that all seven integration
       tests exercise a single, consistent artifact contract.
"""

def create_integration_data(n=1000, seed=42):
    """
    Create synthetic test data for integration testing.

    Args:
        n: Number of samples
        seed: Random seed for reproducibility

    Returns:
        DataFrame with OHLCV, feature columns, state, and a deterministic
        timestamp column required by Module 108.
    """
    rng = np.random.default_rng(seed)

    # Generate price data
    close = 100 + np.cumsum(rng.normal(0, 0.5, n))
    high = close + np.abs(rng.normal(0, 1.0, n))
    low = close - np.abs(rng.normal(0, 1.0, n))
    open_price = close - rng.normal(0, 0.3, n)

    # Generate feature columns for Module 109 to detect
    returns = pd.Series(close).pct_change()
    volatility = returns.rolling(10).std()
    momentum = pd.Series(close).pct_change(5)
    rsi = 50 + rng.normal(0, 10, n)
    volume = rng.uniform(1000, 10000, n)

    df = pd.DataFrame({
        "open": open_price,
        "high": high,
        "low": low,
        "close": close,
        "volume": volume,
        "return_1": returns,
        "return_5": pd.Series(close).pct_change(5),
        "volatility": volatility,
        "momentum": momentum,
        "rsi": rsi,
        "state": rng.integers(0, 16, n),
    })

    # Drop NaN rows from feature creation and reset index so timestamps
    # align 1:1 with the surviving rows.
    df = df.dropna().reset_index(drop=True)

    # Deterministic market timestamps are REQUIRED by Module 108.
    # They must exist before Module 108 computes forward targets so that
    # row-level breakout labels cannot be mis-aligned across the 108→109
    # handoff. Never synthesize timestamps in production code — only here.
    df.insert(
        0,
        "timestamp",
        pd.date_range(
            "2020-01-01",
            periods=len(df),
            freq="15min",
        ),
    )

    return df


def test_direct_adapter_path():
    """
    Test 1: Direct adapter path
    108 → target_adapter → 109
    """
    print("\n" + "=" * 60)
    print("TEST 1: Direct Adapter Path")
    print("108 → target_adapter → 109")
    print("=" * 60)

    df = create_integration_data(n=500)
    assert "timestamp" in df.columns, "Synthetic integration data must carry timestamps"

    print("  Step 1: Running Module 108 (Breakout Probability)...")

    # Horizons must include the canonical 10-bar target so that the
    # resulting artifact exposes the full per-horizon breakout set
    # (including breakout_5) rather than only a single canonical column.
    module108 = BreakoutProbabilityResearch(
        horizons=[1, 3, 5, 10, 20],
        atr_period=14,
        k_multiplier=1.0,
        min_samples=10,
    )
    result108 = module108.run_dataframe(df, dataset_id="integration_test_108")

    assert result108.status == "completed"
    assert "features" in result108.artifacts
    print("  ✓ Module 108 completed successfully")

    targets = result108.artifacts["features"]
    assert "breakout_5" in targets.columns
    print(
        "  ✓ Module 108 produced breakout targets: "
        f"{[c for c in targets.columns if c.startswith('breakout_')]}"
    )

    print("  Step 2: Attaching breakout target via adapter...")
    df109 = attach_breakout_target(
        df=df,
        breakout_targets=targets,
        horizon=5,
        target_column="future_breakout"
    )

    assert "future_breakout" in df109.columns
    print("  ✓ Target column 'future_breakout' attached")

    valid_targets = df109["future_breakout"].dropna()
    assert np.all(np.isin(valid_targets.unique(), [-99, -1, 0, 1]))
    print(f"  ✓ Target values valid: {valid_targets.unique()}")

    breakout_cols = [c for c in df109.columns if c.startswith("breakout_")]
    print(f"  ✓ breakout_* columns present: {breakout_cols}")

    print("  Step 3: Running Module 109 (Feature Importance)...")
    module109 = FeatureImportanceResearch(
        target_column="future_breakout",
        target_horizon=5,
        n_permutations=3,
        n_bootstrap=10,
        n_folds=3,
        random_state=42,
        include_state=False
    )
    result109 = module109.run_dataframe(df109, dataset_id="integration_test_109")

    assert result109.status == "completed"
    print("  ✓ Module 109 completed successfully")

    print("  Step 4: Verifying feature leakage protection...")
    engine = FeatureImportanceEngine(
        target_column="future_breakout",
        include_state=False
    )
    detected_features = engine.detect_features(df109)

    breakout_features_detected = [f for f in detected_features if f.startswith("breakout_")]
    assert len(breakout_features_detected) == 0, \
        f"Found breakout_* features: {breakout_features_detected}"
    print(
        f"  ✓ Detected features: {detected_features[:5]}"
        f"{'...' if len(detected_features) > 5 else ''}"
    )
    print("  ✓ No breakout_* columns detected as features")

    feature_ranking = result109.artifacts["feature_ranking"]
    breakout_in_ranking = [f for f in feature_ranking.index if f.startswith("breakout_")]
    assert len(breakout_in_ranking) == 0, \
        f"Found breakout_* in ranking: {breakout_in_ranking}"
    print("  ✓ Feature ranking does not contain breakout_* columns")

    assert result109.statistics.get("validation_passed") == True
    print("  ✓ Validation passed")

    print("  ✓ Direct adapter path test passed")
    return True


def test_dependency_chain_path():
    """
    Test 2: Full dependency path
    ResearchRunner → ModuleLoader → 103 → 108 → ArtifactStore
    → DependencyInjector → 109
    """
    print("\n" + "=" * 60)
    print("TEST 2: Full Dependency Chain Path")
    print(
        "ResearchRunner → ModuleLoader → 103 → 108 → ArtifactStore "
        "→ DependencyInjector → 109"
    )
    print("=" * 60)

    print("  Step 1: Running Module 108 (Breakout Probability)...")

        horizons=[1, 3, 5, 10, 20],
        atr_period=14,

    print("  Step 2: Simulating ArtifactStore → DependencyInjector...")

    assert len(target_cols) > 0
    print(f"  ✓ Created target columns: {target_cols}")

    print("  Step 3: Running Module 109 with injected dependency...")
    module109 = FeatureImportanceResearch(
        target_column="future_breakout_5",
        target_horizon=5,
        n_permutations=3,
        n_bootstrap=10,
        n_folds=3,
        random_state=42,
        include_state=False
    )
    result109 = module109.run_dataframe(df109, dataset_id="dependency_test_109")

    assert result109.status == "completed"
    print("  ✓ Module 109 completed")

    metadata = FeatureImportanceResearch.METADATA
    assert "depends_on" in metadata
    assert "108" in metadata["depends_on"]
    print("  ✓ Module 109 metadata declares dependency on Module 108")

    feature_ranking = result109.artifacts["feature_ranking"]
    breakout_in_ranking = [f for f in feature_ranking.index if f.startswith("breakout_")]
    assert len(breakout_in_ranking) == 0
    print("  ✓ No breakout_* features in ranking")

    assert result109.statistics.get("validation_passed") == True
    print("  ✓ Validation passed")

    print("  ✓ Full dependency chain test passed")

def test_feature_leakage_protection():
    """
    Test 3: Feature leakage protection (explicit test).
    Verify that breakout_* columns are never selected as features.
    """
    print("\n" + "=" * 60)
    print("TEST 3: Feature Leakage Protection")
    print("=" * 60)

    df = create_integration_data(n=200)

    df_with_breakout = df.copy()
    df_with_breakout["breakout_1"] = np.random.choice([-1, 0, 1], len(df))
    df_with_breakout["breakout_3"] = np.random.choice([-1, 0, 1], len(df))
    df_with_breakout["breakout_5"] = np.random.choice([-1, 0, 1], len(df))
    df_with_breakout["breakout_10"] = np.random.choice([-1, 0, 1], len(df))
    df_with_breakout["future_breakout"] = np.random.choice([-1, 0, 1], len(df))

    print("  Test 1: Auto-detection exclusion")
    engine = FeatureImportanceEngine(
        target_column="future_breakout",
        include_state=False
    )
    detected_features = engine.detect_features(df_with_breakout)

    assert "breakout_1" not in detected_features
    assert "breakout_3" not in detected_features
    assert "breakout_5" not in detected_features
    assert "breakout_10" not in detected_features
    print("  ✓ Auto-detection excludes all breakout_* columns")
    print(f"    Detected {len(detected_features)} features (breakout_* excluded)")

    print("  Test 2: Explicit feature_columns protection")
    engine_explicit = FeatureImportanceEngine(
        target_column="future_breakout",
        feature_columns=["return_1", "volatility", "breakout_5", "momentum"],
        include_state=False
    )
    try:
        engine_explicit.prepare_data(df_with_breakout)
        assert False, "Should have raised ValueError"
    except ValueError as e:
        print(f"    Caught expected error: {e}")
        assert "Future-derived Module 108 target columns cannot be used" in str(e)
        print("  ✓ Explicit feature_columns protection works")

    print("  Test 3: Full pipeline protection")
    module109 = FeatureImportanceResearch(
        target_column="future_breakout",
        target_horizon=5,
        n_permutations=3,
        n_bootstrap=5,
        n_folds=2,
        random_state=42,
        include_state=False
    )
    result109 = module109.run_dataframe(df_with_breakout, dataset_id="leakage_test")

    assert result109.status == "completed"

    feature_ranking = result109.artifacts["feature_ranking"]
    breakout_in_ranking = [f for f in feature_ranking.index if f.startswith("breakout_")]
    assert len(breakout_in_ranking) == 0
    print("  ✓ Full pipeline excludes breakout_* features")

    print("  ✓ Feature leakage protection test passed")
    return True
