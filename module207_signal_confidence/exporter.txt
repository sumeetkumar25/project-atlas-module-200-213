"""
Module 207 - Signal Confidence Exporter
"""

import pandas as pd
from pathlib import Path


class SignalConfidenceExporter:
    """Exporter for signal confidence results."""
    
    @staticmethod
    def export_confidence_matrix(df: pd.DataFrame, filepath: str):
        """Export confidence matrix to Parquet."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df.to_parquet(path, index=False)
    
    @staticmethod
    def export_signal_confidence(df: pd.DataFrame, filepath: str):
        """Export signal confidence to CSV."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df.to_csv(path, index=False)
    
    @staticmethod
    def export_confidence_statistics(df: pd.DataFrame, filepath: str):
        """Export confidence statistics to CSV."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df.to_csv(path, index=False)
    
    @staticmethod
    def export_confidence_distribution(df: pd.DataFrame, filepath: str):
        """Export confidence distribution to CSV."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df.to_csv(path, index=False)
