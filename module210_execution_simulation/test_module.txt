def create_test_data(n=100, seed=42):
    """Create synthetic test data for execution simulation."""
    np.random.seed(seed)
    
    timestamps = pd.date_range('2020-01-01', periods=n, freq='1h')
    
    # Create portfolio data with fractional positions
    approved_position = np.random.choice(
        [-1.0, 0.0, 1.0],
        n,
        p=[0.2, 0.6, 0.2]
    ).astype(float)
    
    approved_size = np.abs(approved_position) * np.random.uniform(0.2, 0.8, n)
    risk_state = np.random.choice(['APPROVED', 'REDUCED', 'REJECTED', 'FLAT'], n)
    
    portfolio_df = pd.DataFrame({
        'timestamp': timestamps,
        'approved_position': approved_position,
        'approved_size': approved_size,
        'risk_state': risk_state,
        'signal': np.random.randn(n) * 0.3,
        'confidence': np.random.uniform(0.4, 0.9, n),
        'open': 100 + np.random.randn(n) * 0.5,
        'high': 100 + np.random.randn(n) * 0.5 + np.abs(np.random.randn(n) * 0.2),
        'low': 100 + np.random.randn(n) * 0.5 - np.abs(np.random.randn(n) * 0.2),
        'close': 100 + np.random.randn(n) * 0.5
    })
    
    return {
        'portfolio_matrix': portfolio_df
    }


def test_engine_load_portfolio():
    """Test 1: Load portfolio matrix."""
    print("\n" + "=" * 60)
    print("TEST 1: Load Portfolio Matrix")
    print("=" * 60)
    
    data = create_test_data()
    engine = ExecutionSimulationEngine()
    
    with tempfile.TemporaryDirectory() as tmpdir:
        tmp_path = Path(tmpdir)
        matrix_path = tmp_path / "portfolio_matrix.parquet"
        data['portfolio_matrix'].to_parquet(matrix_path, index=False)
        
        loaded_df = engine.load_portfolio(matrix_path)
        
        assert not loaded_df.empty
        assert 'timestamp' in loaded_df.columns
        assert 'approved_position' in loaded_df.columns
    
    print("  ✓ Load portfolio matrix test passed")
    return True

def test_engine_execution_delay():
    """Test 3: Execution delay."""
    print("\n" + "=" * 60)
    print("TEST 3: Execution Delay")
    print("=" * 60)
    
    config = ExecutionConfig(latency_bars=1)
    engine = ExecutionSimulationEngine(config=config)
    
    data = create_test_data()
    portfolio_df = data['portfolio_matrix']
    orders_df = engine.generate_orders(portfolio_df)
    
    delayed_orders = engine.apply_execution_delay(orders_df, portfolio_df)
    
    if not delayed_orders.empty:
        assert 'execution_index' in delayed_orders.columns
        assert 'execution_timestamp' in delayed_orders.columns
        # Check delay applied
        for _, order in delayed_orders.iterrows():
            if order['order_index'] + config.latency_bars < len(portfolio_df):
                assert order['execution_index'] == order['order_index'] + config.latency_bars
    
    print("  ✓ Execution delay test passed")
    return True


def test_engine_slippage():
    """Test 4: Slippage."""
    print("\n" + "=" * 60)
    print("TEST 4: Slippage")
    print("=" * 60)
    
    config = ExecutionConfig(slippage_rate=0.0001)
    engine = ExecutionSimulationEngine(config=config)
    
    data = create_test_data()
    portfolio_df = data['portfolio_matrix']
    
    # Test buy order (positive slippage)
    price, slippage = engine.apply_slippage(portfolio_df, 0, "BUY")
    assert slippage >= 0
    assert price >= portfolio_df['open'].iloc[0]
    
    # Test sell order (negative slippage)
    price, slippage = engine.apply_slippage(portfolio_df, 0, "SELL")
    assert slippage >= 0
    assert price <= portfolio_df['open'].iloc[0]
    
    print("  ✓ Slippage test passed")
    return True

    portfolio_df.loc[0, 'approved_position'] = 0.0
    portfolio_df.loc[1, 'approved_position'] = 0.5

    orders_df = engine.generate_orders(portfolio_df)
    delayed_orders = engine.apply_execution_delay(orders_df, portfolio_df)
    execution_events, position_history = engine.execute_orders(portfolio_df, delayed_orders)
    trade_log = engine.build_trade_log(execution_events, portfolio_df)
    
    # Should have created trades
    if not trade_log.empty:
        # Check trade fields
        assert 'trade_id' in trade_log.columns
        assert 'direction' in trade_log.columns
        assert 'entry_price' in trade_log.columns
        assert 'exit_price' in trade_log.columns
        assert 'executed_size' in trade_log.columns
        
        # Check fractional positions
        for _, trade in trade_log.iterrows():
            assert trade['executed_size'] > 0
            # Size should be a float (0.5, 0.3, 0.75, etc.)
