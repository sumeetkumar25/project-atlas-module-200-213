"""
Module 205 - Feature Selection Report
"""

from datetime import datetime
import pandas as pd


class FeatureSelectionReport:
    """Report generator for feature selection results."""
    
    @staticmethod
    def generate(statistics, metadata=None, dataset_name="Unknown") -> str:
        """Generate a formatted report."""
        lines = []
        
        lines.append("=" * 55)
        lines.append("MODULE 205")
        lines.append("FEATURE SELECTION & RANKING")
        lines.append("=" * 55)
        lines.append(f"\nDataset: {dataset_name}")
        
        # Overview
        lines.append("\n" + "-" * 45)
        lines.append("OVERVIEW")
        lines.append("-" * 45)
        lines.append(f"  Total Features: {statistics.total_features}")
        lines.append(f"  SELECTED: {statistics.selected_count}")
        lines.append(f"  REJECTED: {statistics.rejected_count}")
        
        # Percentages
        lines.append("\n" + "-" * 45)
        lines.append("PERCENTAGES")
        lines.append("-" * 45)
        lines.append(f"  SELECTED: {statistics.selected_percentage:.1f}%")
        lines.append(f"  REJECTED: {statistics.rejected_percentage:.1f}%")
        
        # Scores
        lines.append("\n" + "-" * 45)
        lines.append("SCORES")
        lines.append("-" * 45)
        lines.append(f"  Average: {statistics.average_score:.3f}")
        lines.append(f"  Median: {statistics.median_score:.3f}")
        lines.append(f"  Best: {statistics.best_score:.3f}")
        lines.append(f"  Worst: {statistics.worst_score:.3f}")
        
        # Best and Worst Features
        if statistics.best_feature:
            lines.append("\n" + "-" * 45)
            lines.append("BEST FEATURE")
            lines.append("-" * 45)
            lines.append(f"  {statistics.best_feature}: {statistics.best_score:.3f}")
        
        if statistics.worst_feature:
            lines.append("\n" + "-" * 45)
            lines.append("WORST FEATURE")
            lines.append("-" * 45)
            lines.append(f"  {statistics.worst_feature}: {statistics.worst_score:.3f}")
        
        # Thresholds
        lines.append("\n" + "-" * 45)
        lines.append("THRESHOLDS")
        lines.append("-" * 45)
        lines.append(f"  Predictive Threshold: {statistics.selection_threshold}")
        lines.append(f"  Stability Threshold: {statistics.stability_threshold}")
        
        # Interpretation
        lines.append("\n" + "-" * 45)
        lines.append("INTERPRETATION")
        lines.append("-" * 45)
        
        if statistics.average_score > 0.6:
            lines.append("  ✅ Features show strong overall quality")
        elif statistics.average_score > 0.4:
            lines.append("  ✓ Features show moderate overall quality")
        else:
            lines.append("  ⚠ Features show weak overall quality")
        
        if statistics.selected_percentage > 50:
            lines.append("  ✓ Most features passed selection")
        else:
            lines.append("  ⚠ Significant feature reduction applied")
        
        # Validation
        lines.append("\n" + "-" * 45)
        lines.append("VALIDATION")
        lines.append("-" * 45)
        lines.append("  Status: PASSED")
        
        # Footer
        lines.append(f"\n  Generated: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
        lines.append("\n" + "=" * 55)
        lines.append("END OF REPORT")
        lines.append("=" * 55)
        
        return "\n".join(lines)
    
    @staticmethod
    def generate_ranking_table(feature_ranking, top_n=10):
        """
        Generate a ranking table report.

        Displays BOTH raw and normalized scores so the reader can
        distinguish between:

            - Raw scores: absolute quality values used for hard
              filter thresholds (predictive >= 0.40, stability >= 0.50)

            - Normalized scores: relative ranking values within this
              batch, rescaled to [0, 1]. A normalized stability of
              0.0 does NOT mean zero stability — it means "lowest
              stability among the current candidates."

        Without both columns visible, a reader could easily
        misinterpret normalized 0.0 as an absolute failure.
        """
        lines = []
        lines.append("\n" + "=" * 90)
        lines.append(f"TOP {top_n} FEATURES (raw + normalized)")
        lines.append("=" * 90)

        # ------------------------------------------------------------
        # Copy the caller's DataFrame so we can safely add or mutate
        # columns without touching the original.
        # ------------------------------------------------------------
        ranking_df = feature_ranking.copy()

        if ranking_df.empty:
            return "No feature ranking available."

        # ------------------------------------------------------------
        # Defensive: ensure a 'predictive_score' column exists before
        # rendering. Fall back to 'final_score' or 'score' if needed,
        # so the ranking table renders correctly even if the ranking
        # DataFrame was produced by an older engine or a custom pipeline.
        # ------------------------------------------------------------
        if "predictive_score" not in ranking_df.columns:
            if "final_score" in ranking_df.columns:
                ranking_df["predictive_score"] = ranking_df["final_score"]
            elif "score" in ranking_df.columns:
                ranking_df["predictive_score"] = ranking_df["score"]
            else:
                raise ValueError(
                    "Feature ranking is missing predictive_score, "
                    "final_score, and score."
                )

        # ------------------------------------------------------------
        # Defensive: ensure the raw columns exist. If they are absent
        # (older engine, custom pipeline), fall back to the normalized
        # columns so the table still renders — but emit a warning
        # marker in the header so the reader knows the raw columns
        # were not available.
        # ------------------------------------------------------------
        raw_columns_available = (
            "raw_predictive_score" in ranking_df.columns and
            "raw_stability_score" in ranking_df.columns
        )

        if not raw_columns_available:
            if "raw_predictive_score" not in ranking_df.columns:
                ranking_df["raw_predictive_score"] = ranking_df.get(
                    "predictive_score", 0.0
                )
            if "raw_stability_score" not in ranking_df.columns:
                ranking_df["raw_stability_score"] = ranking_df.get(
                    "stability_score", 0.0
                )

        # ------------------------------------------------------------
        # Sort by the resolved predictive score and take the top N.
        # ------------------------------------------------------------
        ranking_df = ranking_df.sort_values(
            "predictive_score",
            ascending=False
        ).head(top_n)

        if not raw_columns_available:
            lines.append(
                "  NOTE: raw score columns missing; "
                "displayed raw values fall back to normalized."
            )

        # ------------------------------------------------------------
        # Header — explicitly label both families so the reader
        # understands the semantic difference.
        # ------------------------------------------------------------
        lines.append(
            "\n"
            f"{'Rank':<6}"
            f"{'Feature':<24}"
            f"{'Final':<9}"
            f"{'Pred(norm)':<12}"
            f"{'Pred(raw)':<12}"
            f"{'Stab(norm)':<12}"
            f"{'Stab(raw)':<12}"
        )
        lines.append("-" * 90)

        for _, row in ranking_df.iterrows():
            lines.append(
                f"{row['rank']:<6}"
                f"{str(row['feature'])[:23]:<24}"
                f"{row['final_score']:<9.3f}"
                f"{row['predictive_score']:<12.3f}"
                f"{row['raw_predictive_score']:<12.3f}"
                f"{row['stability_score']:<12.3f}"
                f"{row['raw_stability_score']:<12.3f}"
            )

        lines.append("")
        lines.append("-" * 90)
        lines.append(
            "  Legend:"
        )
        lines.append(
            "    Final        = weighted combination (clipped to [0,1])"
        )
        lines.append(
            "    Pred(norm)   = predictive score rescaled to [0,1] within this batch"
        )
        lines.append(
            "    Pred(raw)    = predictive score used for hard threshold (>=0.40)"
        )
        lines.append(
            "    Stab(norm)   = stability score rescaled to [0,1] within this batch"
        )
        lines.append(
            "    Stab(raw)    = stability score used for hard threshold (>=0.50)"
        )
        lines.append(
            "  A normalized value of 0.0 means 'lowest in this batch',"
        )
        lines.append(
            "  NOT 'zero absolute quality'. Consult the raw column."
        )
        lines.append("=" * 90)

        return "\n".join(lines)
    
    @staticmethod
    def save(report: str, filepath: str):
        """Save report to file."""
        from pathlib import Path
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        with open(path, "w", encoding="utf-8") as f:
            f.write(report)
