"""
Module 205 - Feature Selection & Ranking Engine

Produces the final feature set that every future model in Project Atlas will use.

Inputs:
  - Module203: predictive_feature_matrix.parquet
  - Module202: feature_stability.csv
  - Module203: interaction_statistics.csv
  - Module204: feature_redundancy.csv, feature_clusters.csv, removed_features.csv

Outputs:
  - Selected features with final scores
  - Feature rankings
  - Selection statistics
"""

import numpy as np
import pandas as pd
from dataclasses import dataclass
from typing import Optional, Dict, Any, List, Tuple
from pathlib import Path
import warnings
warnings.filterwarnings('ignore')


@dataclass
class FeatureSelectionResult:
    """Results from feature selection."""
    selected_features: List[str]
    rejected_features: List[str]
    feature_scores: pd.DataFrame
    feature_ranking: pd.DataFrame
    summary_statistics: Dict[str, Any]


class FeatureSelectionEngine:
    """Core engine for feature selection and ranking."""

    # ------------------------------------------------------------------
    # Scoring weights
    #
    # The interaction weight is 0.00 because Module 203 does not
    # currently supply an independent interaction statistic. Reusing
    # Module 203's predictive_score as the interaction score would
    # double-count the same evidence under two different weights.
    #
    # The weight mass freed by zeroing interaction is redistributed
    # to the two metrics that represent genuine, independent evidence:
    #   - predictive (Module 203 predictive evidence)
    #   - stability  (Module 202 stability evidence)
    #
    # The weights are not required to sum to 1 because the final
    # score is clipped to [0, 1] after combination.
    # ------------------------------------------------------------------
    SCORING_WEIGHTS = {
        "predictive": 0.45,
        "stability": 0.30,
        "interaction": 0.00,
        "synergy": 0.15,
        "redundancy": -0.10
    }

    # Hard thresholds for automatic rejection.
    #
    # These thresholds are applied to RAW scores (not normalized
    # scores) so that they represent absolute quality floors,
    # independent of the distribution of any single batch.
    PREDICTIVE_THRESHOLD = 0.40
    STABILITY_THRESHOLD = 0.50

    def __init__(
        self,
        predictive_threshold: float = 0.40,
        stability_threshold: float = 0.50,
        timestamp_column: str = "timestamp"
    ):
        """
        Initialize the feature selection engine.
        
        Args:
            predictive_threshold: Minimum raw predictive score to keep
            stability_threshold: Minimum raw stability score to keep
            timestamp_column: Name of the timestamp column
        """
        self.predictive_threshold = predictive_threshold
        self.stability_threshold = stability_threshold
        self.timestamp_column = timestamp_column

    def load_predictive_matrix(self, matrix_path: Path) -> pd.DataFrame:
        """Load the predictive feature matrix."""
        if not matrix_path.exists():
            raise FileNotFoundError(f"Predictive matrix not found: {matrix_path}")
        
        if matrix_path.suffix == '.parquet':
            df = pd.read_parquet(matrix_path)
        else:
            df = pd.read_csv(matrix_path)
            if self.timestamp_column in df.columns:
                df[self.timestamp_column] = pd.to_datetime(df[self.timestamp_column])
        
        return df

    def load_stability_scores(self, filepath: Path) -> pd.DataFrame:
        """Load stability scores from Module 202."""
        if not filepath.exists():
            raise FileNotFoundError(f"Stability scores not found: {filepath}")
        
        df = pd.read_csv(filepath)
        return df

    def load_interaction_scores(self, filepath: Path) -> pd.DataFrame:
        """Load interaction statistics from Module 203."""
        if not filepath.exists():
            raise FileNotFoundError(f"Interaction scores not found: {filepath}")
        
        df = pd.read_csv(filepath)
        return df

    def load_redundancy_data(self, filepath: Path) -> pd.DataFrame:
        """Load redundancy data from Module 204."""
        if not filepath.exists():
            raise FileNotFoundError(f"Redundancy data not found: {filepath}")
        
        df = pd.read_csv(filepath)
        return df

    def load_removed_features(self, filepath: Path) -> List[str]:
        """Load removed features from Module 204."""
        if not filepath.exists():
            return []
        
        df = pd.read_csv(filepath)
        if 'removed_features' in df.columns:
            return df['removed_features'].tolist()
        return df.iloc[:, 0].tolist()

    def get_feature_columns(self, df: pd.DataFrame) -> List[str]:
        """Get feature columns (exclude timestamp and OHLCV)."""
        ignore_cols = {
            "timestamp", "date", "time", "index",
            "close", "open", "high", "low", "volume"
        }
        return [
            column for column in df.columns
            if column not in ignore_cols
        ]

    def normalize_scores(self, scores: pd.Series, min_val: float = 0, max_val: float = 1) -> pd.Series:
        """Normalize scores to [0, 1] range."""
        if scores.min() == scores.max():
            return pd.Series(0.5, index=scores.index)
        return (scores - scores.min()) / (scores.max() - scores.min())

    def compute_final_scores(
        self,
        features: List[str],
        predictive_scores: pd.Series,
        stability_scores: pd.Series,
        interaction_scores: pd.Series,
        synergy_scores: pd.Series,
        redundancy_scores: pd.Series
    ) -> pd.DataFrame:
        """
        Compute final scores using weighted combination.

        The returned DataFrame contains TWO families of columns:

        1. Normalized columns (used for the weighted ranking):
              predictive_score, stability_score,
              interaction_score, synergy_score,
              redundancy_score

        2. Raw columns (used for absolute hard filters):
              raw_predictive_score, raw_stability_score

        The distinction matters: normalization rescales values to
        [0, 1] based on the batch's own min/max, which makes the
        thresholds meaningless as absolute quality floors. The raw
        columns preserve the original scores so that
        PREDICTIVE_THRESHOLD and STABILITY_THRESHOLD can be applied
        consistently across every batch.
        """
        # Normalize each metric to [0, 1]
        pred_norm = self.normalize_scores(predictive_scores)
        stab_norm = self.normalize_scores(stability_scores)
        inter_norm = self.normalize_scores(interaction_scores)
        syn_norm = self.normalize_scores(synergy_scores)
        red_norm = self.normalize_scores(redundancy_scores)
        
        # Combine with weights
        final_scores = (
            self.SCORING_WEIGHTS["predictive"] * pred_norm +
            self.SCORING_WEIGHTS["stability"] * stab_norm +
            self.SCORING_WEIGHTS["interaction"] * inter_norm +
            self.SCORING_WEIGHTS["synergy"] * syn_norm +
            self.SCORING_WEIGHTS["redundancy"] * red_norm
        )
        
        # Clip to [0, 1]
        final_scores = np.clip(final_scores, 0.0, 1.0)
        
        # Create DataFrame
        result_df = pd.DataFrame({
            "feature": features,

            # Normalized values are used for weighted ranking.
            "predictive_score": pred_norm.values,
            "stability_score": stab_norm.values,
            "interaction_score": inter_norm.values,
            "synergy_score": syn_norm.values,
            "redundancy_score": red_norm.values,

            # Raw values are used for absolute hard filters.
            "raw_predictive_score": predictive_scores.values,
            "raw_stability_score": stability_scores.values,

            "final_score": final_scores.values,
        })
        
        return result_df

    def apply_hard_filters(
        self,
        scores_df: pd.DataFrame,
        removed_features: List[str],
        all_features: List[str],
        module204_removed_features: Optional[List[str]] = None
    ) -> Tuple[pd.DataFrame, List[str], List[str]]:
        """
        Apply hard filters to reject features that fail minimum thresholds.

        The thresholds are applied to the RAW score columns
        (raw_predictive_score, raw_stability_score), not the
        normalized ones. This ensures the thresholds represent
        absolute quality floors rather than batch-relative ranks.

        Module 204's 'removed_features' list is treated as ADVISORY,
        not authoritative. A feature that Module 204 flagged for
        redundancy removal is still retained when it independently
        passes both the predictive and stability thresholds. This
        prevents Module 204's redundancy view from silently
        discarding features that are demonstrably useful on their
        own merits.

        Args:
            scores_df: Feature scores DataFrame
            removed_features: Features flagged for removal by Module 204
            all_features: All candidate features from Module 203
            module204_removed_features: Optional explicit list of
                features removed by Module 204. Currently unused for
                trust decisions, but retained for API compatibility
                and future diagnostic reporting.
        """
        # Start with all features
        selected = all_features.copy()
        rejected = []

        # Check each feature
        for feature in all_features:
            row = scores_df[scores_df['feature'] == feature]
            if row.empty:
                rejected.append(feature)
                continue

            # --------------------------------------------------------
            # Apply thresholds to RAW scores.
            #
            # The raw columns are guaranteed to exist because
            # compute_final_scores() always produces them. If they
            # are missing (schema drift), the .get-based fallback
            # resolves to the normalized score columns so that the
            # pipeline still functions — but the fix is to ensure
            # compute_final_scores() emits the raw columns.
            # --------------------------------------------------------
            predictive = row["raw_predictive_score"].iloc[0]
            stability = row["raw_stability_score"].iloc[0]

            # Remove if predictive is too low
            if predictive < self.predictive_threshold:
                rejected.append(feature)
                selected.remove(feature)
                continue

            # Remove if stability is too low
            if stability < self.stability_threshold:
                rejected.append(feature)
                selected.remove(feature)
                continue

            # --------------------------------------------------------
            # Module 204 redundancy removal is advisory.
            #
            # Do not reject a feature that independently passes
            # predictive and stability requirements. Module 204's
            # view of redundancy is pairwise and context-dependent;
            # a feature can be highly correlated with a sibling yet
            # still carry unique predictive signal that the
            # composite score undervalues.
            # --------------------------------------------------------
            if feature in removed_features:
                # Module 204 redundancy removal is advisory.
                # Do not reject a feature that independently passes
                # predictive and stability requirements.
                if predictive >= self.predictive_threshold and stability >= self.stability_threshold:
                    continue

                rejected.append(feature)
                selected.remove(feature)
                continue

        # Filter DataFrame
        filtered_df = scores_df[scores_df['feature'].isin(selected)]

        return filtered_df, selected, rejected

    def rank_features(self, scores_df: pd.DataFrame) -> pd.DataFrame:
        """
        Rank features by predictive score.

        IMPORTANT:
        - The ranking MUST include every scored feature, not only the
          selected subset. A feature that was rejected by the hard
          filters is still part of the audit trail, and its predictive
          score is still meaningful for downstream reporting.
        - The ranking MUST contain a 'predictive_score' column. The
          score column is resolved with a fallback chain so that the
          ranking is robust to schema drift in the upstream scores
          DataFrame:
              predictive_score → final_score → score
          If none of these exist, the method raises KeyError.
        """

        feature_ranking = scores_df.copy()

        if "predictive_score" not in feature_ranking.columns:
            if "final_score" in feature_ranking.columns:
                feature_ranking["predictive_score"] = feature_ranking["final_score"]
            elif "score" in feature_ranking.columns:
                feature_ranking["predictive_score"] = feature_ranking["score"]
            else:
                raise KeyError(
                    "Module 205 cannot create ranking: "
                    "no predictive_score, final_score, or score column exists"
                )

        feature_ranking["rank"] = (
            feature_ranking["predictive_score"]
            .rank(method="first", ascending=False)
            .astype(int)
        )

        feature_ranking = feature_ranking.sort_values(
            "rank"
        ).reset_index(drop=True)

        return feature_ranking

    def compute_statistics(
        self,
        all_scores_df: pd.DataFrame,
        selected: List[str],
        rejected: List[str]
    ) -> Dict[str, Any]:
        """
        Compute summary statistics using the full scores DataFrame (before filtering).
        """
        total = len(all_scores_df)
        scores = all_scores_df['final_score']

        # Handle valid zero-feature research outcome
        if total == 0:
            return {
                'total_features': 0,
                'selected_count': len(selected),
                'rejected_count': len(rejected),
                'selected_percentage': 0.0,
                'rejected_percentage': 0.0,
                'average_score': 0.0,
                'median_score': 0.0,
                'min_score': 0.0,
                'max_score': 0.0,
                'best_feature': None,
                'worst_feature': None,
                'best_score': 0.0,
                'worst_score': 0.0,
                'selection_threshold': self.predictive_threshold,
                'stability_threshold': self.stability_threshold
            }

        # Find best and worst from all scores
        best_idx = scores.idxmax()
        worst_idx = scores.idxmin()
        
        stats = {
            'total_features': total,
            'selected_count': len(selected),
            'rejected_count': len(rejected),
            'selected_percentage': len(selected) / total * 100 if total > 0 else 0,
            'rejected_percentage': len(rejected) / total * 100 if total > 0 else 0,
            'average_score': float(scores.mean()),
            'median_score': float(scores.median()),
            'min_score': float(scores.min()),
            'max_score': float(scores.max()),
            'best_feature': all_scores_df.iloc[best_idx]['feature'] if not all_scores_df.empty else None,
            'worst_feature': all_scores_df.iloc[worst_idx]['feature'] if not all_scores_df.empty else None,
            'best_score': float(scores.max()),
            'worst_score': float(scores.min()),
            'selection_threshold': self.predictive_threshold,
            'stability_threshold': self.stability_threshold
        }
        
        return stats

    def save_outputs(
        self,
        selected: List[str],
        rejected: List[str],
        scores_df: pd.DataFrame,
        ranking_df: pd.DataFrame,
        stats: Dict[str, Any],
        output_dir: Path
    ) -> None:
        """Save all outputs."""
        output_dir.mkdir(parents=True, exist_ok=True)
        
        # Selected features
        selected_df = pd.DataFrame({'selected_features': selected})
        selected_path = output_dir / "selected_features.csv"
        selected_df.to_csv(selected_path, index=False)
        print(f"  Saved selected features to {selected_path}")
        
        # Feature scores
        scores_path = output_dir / "feature_scores.csv"
        scores_df.to_csv(scores_path, index=False)
        print(f"  Saved feature scores to {scores_path}")
        
        # Feature ranking
        ranking_path = output_dir / "feature_ranking.csv"
        ranking_df.to_csv(ranking_path, index=False)
        print(f"  Saved feature ranking to {ranking_path}")
        
        # Statistics
        stats_df = pd.DataFrame([stats])
        stats_path = output_dir / "feature_selection_statistics.csv"
        stats_df.to_csv(stats_path, index=False)
        print(f"  Saved statistics to {stats_path}")

    def compute(
        self,
        predictive_matrix_path: Path,
        stability_file: Path,
        interaction_file: Path,
        redundancy_file: Path,
        removed_file: Path,
        output_dir: Optional[Path] = None
    ) -> FeatureSelectionResult:
        """
        Main computation method.
        """
        print("\n  Computing Feature Selection & Ranking...")
        
        if output_dir is None:
            output_dir = Path("data/feature_store/selected")
        
        # Load all inputs
        print(f"  Loading predictive matrix from: {predictive_matrix_path}")
        predictive_df = self.load_predictive_matrix(predictive_matrix_path)

        # ------------------------------------------------------------
        # Fail clearly when Module 203 produced no usable features.
        #
        # An empty feature set here means Module 203 wrote a matrix
        # containing only timestamp/OHLCV columns. Module 205 has
        # nothing to select from, so it must abort with a message
        # that pinpoints the upstream problem.
        # ------------------------------------------------------------

        features = self.get_feature_columns(predictive_df)

        if not features:
            raise ValueError(
                "Module 205 received no feature columns from Module 203. "
                "Expected timestamp plus one or more predictive features."
            )

        print(f"  Found {len(features)} features")
        
        print(f"  Loading stability scores from: {stability_file}")
        stability_df = self.load_stability_scores(stability_file)
        
        print(f"  Loading interaction scores from: {interaction_file}")
        interaction_df = self.load_interaction_scores(interaction_file)
        
        print(f"  Loading redundancy data from: {redundancy_file}")
        redundancy_df = self.load_redundancy_data(redundancy_file)
        
        print(f"  Loading removed features from: {removed_file}")
        removed_features = self.load_removed_features(removed_file)
        print(f"  Found {len(removed_features)} previously removed features")
        
        # ------------------------------------------------------------
        # Predictive scores: Module 203
        # ------------------------------------------------------------
        if 'predictive_score' in interaction_df.columns:
            predictive_dict = dict(
                zip(interaction_df['feature'], interaction_df['predictive_score'])
            )
            print("  Using predictive scores from Module 203")
        else:
            predictive_dict = {f: 0.5 for f in features}
            print("  Warning: No predictive scores found, using defaults")

        # ------------------------------------------------------------
        # Stability scores: Module 202
        # ------------------------------------------------------------
        stability_dict = dict(
            zip(stability_df['feature'], stability_df['stability_score'])
        )

        # ------------------------------------------------------------
        # Interaction scores
        #
        # IMPORTANT:
        # Module 203 predictive_score is NOT an independent
        # interaction metric. Do not reuse it here because that
        # double-counts the same evidence.
        #
        # Until an independent interaction statistic is supplied,
        # use the neutral value 0.5 for the interaction component.
        #
        # Note: even the neutral 0.5 has no effect on final_score
        # because SCORING_WEIGHTS["interaction"] is 0.00. It is
        # retained here only for schema compatibility so that
        # downstream consumers still see the column.
        # ------------------------------------------------------------
        interaction_dict = {f: 0.5 for f in features}
        print(
            "  No independent interaction metric supplied; "
            "using neutral interaction baseline (0.5)"
        )
        
        # Synergy scores (Module 204 - conditional MI)
        if 'conditional_mi' in redundancy_df.columns:
            synergy_dict = {}
            for feature in features:
                rows = redundancy_df[
                    (redundancy_df['feature_a'] == feature) | 
                    (redundancy_df['feature_b'] == feature)
                ]
                if not rows.empty:
                    synergy_dict[feature] = rows['conditional_mi'].mean()
                else:
                    synergy_dict[feature] = 0.5
        else:
            synergy_dict = {f: 0.5 for f in features}
        
        # Redundancy scores (Module 204 - composite score)
        if 'composite_score' in redundancy_df.columns:
            redundancy_dict = {}
            for feature in features:
                rows = redundancy_df[
                    (redundancy_df['feature_a'] == feature) | 
                    (redundancy_df['feature_b'] == feature)
                ]
                if not rows.empty:
                    redundancy_dict[feature] = rows['composite_score'].mean()
                else:
                    redundancy_dict[feature] = 0.0
        else:
            redundancy_dict = {f: 0.0 for f in features}
        
        # Create series (with proper defaults for missing features)
        predictive_scores = pd.Series({f: predictive_dict.get(f, 0.5) for f in features})
        stability_series = pd.Series({f: stability_dict.get(f, 0.5) for f in features})
        interaction_series = pd.Series({f: interaction_dict.get(f, 0.5) for f in features})
        synergy_series = pd.Series({f: synergy_dict.get(f, 0.5) for f in features})
        redundancy_series = pd.Series({f: redundancy_dict.get(f, 0.0) for f in features})
        
        # Compute final scores
        print("  Computing final scores...")
        scores_df = self.compute_final_scores(
            features,
            predictive_scores,
            stability_series,
            interaction_series,
            synergy_series,
            redundancy_series
        )
        
        # Apply hard filters
        print("  Applying hard filters...")
        filtered_df, selected, rejected = self.apply_hard_filters(
            scores_df,
            removed_features,
            features,
            module204_removed_features=removed_features,
        )
        print(f"    Selected: {len(selected)} features")
        print(f"    Rejected: {len(rejected)} features")
        
        # ------------------------------------------------------------
        # Rank ALL scored features — not just the selected subset.
        #
        # The ranking is an audit artifact and must include every
        # feature that received a final_score, including rejected
        # ones. Filtering the ranking to only `selected` would
        # discard the score evidence for rejected features.
        # ------------------------------------------------------------
        print("  Ranking features...")
        ranking_df = self.rank_features(scores_df)
        
        # Compute statistics (using scores_df which contains ALL features)
        stats = self.compute_statistics(scores_df, selected, rejected)
        
        # Save outputs
        print("\n  Saving outputs...")
        self.save_outputs(selected, rejected, scores_df, ranking_df, stats, output_dir)
        
        print(f"\n  Selection Summary:")
        print(f"    Total Features: {stats['total_features']}")
        print(f"    Selected: {stats['selected_count']}")
        print(f"    Rejected: {stats['rejected_count']}")
        print(f"    Average Score: {stats['average_score']:.3f}")
        print(f"    Best Feature: {stats['best_feature']}")
        print(f"    Worst Feature: {stats['worst_feature']}")
        
        return FeatureSelectionResult(
            selected_features=selected,
            rejected_features=rejected,
            feature_scores=scores_df,  # All features, not just selected
            feature_ranking=ranking_df,
            summary_statistics=stats
        )


# ---------------------------------------------------------------------------
# Module-level entry point
# ---------------------------------------------------------------------------

def run_module205(
    predictive_matrix_path: Path,
    stability_file: Path,
    interaction_file: Path,
    redundancy_file: Path,
    removed_file: Path,
    output_dir: Optional[Path] = None,
    **kwargs
) -> FeatureSelectionResult:
    """
    Convenience function to run Module 205.
    
    Args:
        predictive_matrix_path: Path to predictive feature matrix (Module 203)
        stability_file: Path to feature_stability.csv (Module 202)
        interaction_file: Path to interaction_statistics.csv (Module 203)
        redundancy_file: Path to feature_redundancy.csv (Module 204)
        removed_file: Path to removed_redundant_features.csv (Module 204)
        output_dir: Directory for output artifacts (defaults to data/feature_store/selected)
        **kwargs: Additional arguments passed to FeatureSelectionEngine
        
    Returns:
        FeatureSelectionResult
    """
    engine = FeatureSelectionEngine(**kwargs)
    return engine.compute(
        predictive_matrix_path=predictive_matrix_path,
        stability_file=stability_file,
        interaction_file=interaction_file,
        redundancy_file=redundancy_file,
        removed_file=removed_file,
        output_dir=output_dir,
    )


if __name__ == "__main__":
    import argparse
    
    parser = argparse.ArgumentParser(
        description="Module 205 - Feature Selection & Ranking Engine"
    )
    parser.add_argument("--matrix", type=str, required=True,
                        help="Path to predictive feature matrix (Module 203)")
    parser.add_argument("--stability", type=str, required=True,
                        help="Path to feature_stability.csv (Module 202)")
    parser.add_argument("--interaction", type=str, required=True,
                        help="Path to interaction_statistics.csv (Module 203)")
    parser.add_argument("--redundancy", type=str, required=True,
                        help="Path to feature_redundancy.csv (Module 204)")
    parser.add_argument("--removed", type=str, required=True,
                        help="Path to removed_redundant_features.csv (Module 204)")
    parser.add_argument("--output", type=str, default=None,
                        help="Output directory")
    
    args = parser.parse_args()
    
    result = run_module205(
        predictive_matrix_path=Path(args.matrix),
        stability_file=Path(args.stability),
        interaction_file=Path(args.interaction),
        redundancy_file=Path(args.redundancy),
        removed_file=Path(args.removed),
        output_dir=Path(args.output) if args.output else None,
    )
    
    print("\nDone.")
