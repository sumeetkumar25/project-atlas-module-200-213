# atlas/modules/module205_feature_selection/module.py
"""
Module 205
Feature Selection & Ranking

Produces the final feature set that every future model in Project Atlas will use.

Pipeline: load module outputs → normalize metrics → merge statistics →
          compute score → remove redundant features → rank features →
          validate → export
"""

from typing import Optional, Dict, Any
import pandas as pd
from pathlib import Path

from atlas.core.base_research_module import BaseResearchModule
from atlas.framework.registry import AtlasRegistry

from .engine import FeatureSelectionEngine
from .statistics import FeatureSelectionStatisticsEngine
from .validator import run_validation
from .report import FeatureSelectionReport
from .exporter import FeatureSelectionExporter


class FeatureSelectionResearch(BaseResearchModule):
    """Main research module for feature selection."""
    
    MODULE_OUTPUT = "module205"
    METADATA = FeatureSelectionStatisticsEngine.METADATA
    
    def __init__(
        self,
        registry: Optional[AtlasRegistry] = None,
        predictive_threshold: float = 0.40,
        stability_threshold: float = 0.50,
        timestamp_column: str = "timestamp"
    ):
        """
        Initialize the feature selection module.
        
        Args:
            registry: Optional Atlas registry
            predictive_threshold: Minimum predictive score to keep
            stability_threshold: Minimum stability score to keep
            timestamp_column: Name of the timestamp column
        """
        super().__init__(
            module_name="Feature Selection",
            module_version=self.METADATA["version"],
        )
        
        self.registry = registry or AtlasRegistry()
        self.engine = FeatureSelectionEngine(
            predictive_threshold=predictive_threshold,
            stability_threshold=stability_threshold,
            timestamp_column=timestamp_column
        )
        self.stats_engine = FeatureSelectionStatisticsEngine()
        self.result_data = None
        self.statistics = None
        self.ranking_report = None
        self._output_dir = None
        self._paths = None

    def _register_pipeline_steps(self):
        """Register pipeline steps."""
        self.pipeline_steps = [
            ("feature_selection", self._select_features),
            ("statistics", self._compute_statistics),
            ("validate", self._validate),
            ("export", self._export_results),
        ]
    
    def _prepare_data(self) -> Dict[str, Any]:
        """Prepare data."""
        if self._paths is None:
            raise ValueError("File paths not provided. Use run_selection() method.")
        return self._paths
    
    def _select_features(self) -> Dict[str, Any]:
        """Step 1: Select and rank features."""
        self.logger.info("Step 1: Selecting and ranking features")
        
        prepared = self._prepare_data()
        
        self.result_data = self.engine.compute(
            predictive_matrix_path=prepared["predictive_matrix"],
            stability_file=prepared["stability_file"],
            interaction_file=prepared["interaction_file"],
            redundancy_file=prepared["redundancy_file"],
            removed_file=prepared["removed_file"],
            output_dir=self._output_dir
        )
        
        self.logger.metric(
            "Total Features",
            str(self.result_data.summary_statistics["total_features"])
        )
        self.logger.metric(
            "SELECTED",
            str(self.result_data.summary_statistics["selected_count"])
        )
        self.logger.metric(
            "REJECTED",
            str(self.result_data.summary_statistics["rejected_count"])
        )
        self.logger.metric(
            "Best Feature",
            self.result_data.summary_statistics["best_feature"]
        )
        self.logger.metric(
            "Avg Score",
            f"{self.result_data.summary_statistics['average_score']:.3f}"
        )
        self.logger.success("Feature selection complete")
        
        # Generate report
        self.ranking_report = FeatureSelectionReport.generate_ranking_table(
            self.result_data.feature_ranking,
            top_n=10
        )
        
        return {
            "total_features": self.result_data.summary_statistics["total_features"],
            "selected_count": self.result_data.summary_statistics["selected_count"],
            "rejected_count": self.result_data.summary_statistics["rejected_count"],
            "average_score": self.result_data.summary_statistics["average_score"]
        }
    
    def _compute_statistics(self) -> Dict[str, Any]:
        """Step 2: Computing statistics."""
        self.logger.info("Step 2: Computing statistics")
        self.statistics = self.stats_engine.analyze(self.result_data)
        
        self.logger.metric(
            "Selection Rate",
            f"{self.statistics.selected_percentage:.1f}%"
        )
        self.logger.success("Statistics computed")
        
        return self.stats_engine.summary_dict(self.statistics)
    
    def _validate(self) -> Dict[str, Any]:
        """Step 3: Validation."""
        self.logger.info("Step 3: Validation")
        passed = run_validation(self.result_data, self.statistics)
        self.logger.success("Validation complete")
        return {"software_validation_passed": passed}
    
    def _export_results(self) -> Dict[str, Any]:
        """Step 4: Export."""
        self.logger.info("Step 4: Export")
        
        # Register DataFrames as artifacts
        self.result.artifacts["feature_scores"] = self.result_data.feature_scores
        self.result.artifacts["feature_ranking"] = self.result_data.feature_ranking
        
        # Store selected and rejected features
        selected_df = pd.DataFrame({'selected_features': self.result_data.selected_features})
        rejected_df = pd.DataFrame({'rejected_features': self.result_data.rejected_features})
        
        self.result.artifacts["selected_features"] = selected_df
        self.result.artifacts["rejected_features"] = rejected_df
        
        # Store summary
        summary_df = pd.DataFrame([self.result_data.summary_statistics])
        self.result.artifacts["feature_selection_statistics"] = summary_df
        
        # Generate and register reports
        main_report = FeatureSelectionReport.generate(
            self.statistics,
            metadata=self.METADATA,
            dataset_name=self.result.dataset_identifier,
        )
        self.result.artifacts["feature_selection_report"] = main_report
        self.result.artifacts["ranking_report"] = self.ranking_report
        
        self.logger.metric("Artifacts", str(len(self.result.artifacts)))
        self.logger.success("Export complete")
        
        return {
            "export_status": "complete",
            "artifact_count": len(self.result.artifacts),
            "selected_count": self.result_data.summary_statistics["selected_count"],
            "rejected_count": self.result_data.summary_statistics["rejected_count"]
        }
    
    def compute(self) -> Dict[str, Any]:
        """Run the full computation pipeline."""
        self.logger.section("Feature Selection Pipeline")
        results = self.execute_pipeline()
        self.result.status = "completed"
        self.logger.success("Module 205 complete")
        return results
    
    def run(self, filepath: str, output_dir=None):
        """Run module on file."""
        self._output_dir = Path(output_dir) if output_dir else None
        return super().run(filepath, output_dir)
    
    def run_dataframe(self, df, dataset_id="in_memory", output_dir=None):
        """Run module on a single DataFrame."""
        self._output_dir = Path(output_dir) if output_dir else None
        return super().run_dataframe(df, dataset_id, output_dir)
    
    def run_selection(
        self,
        predictive_matrix: Path,
        stability_file: Path,
        interaction_file: Path,
        redundancy_file: Path,
        removed_file: Path,
        dataset_id: str = "feature_selection",
        output_dir: Optional[Path] = None
    ) -> Any:
        """
        Run the feature selection.
        
        Args:
            predictive_matrix: Path to the predictive feature matrix
            stability_file: Path to the stability scores CSV
            interaction_file: Path to the interaction statistics CSV
            redundancy_file: Path to the redundancy CSV
            removed_file: Path to the removed features CSV
            dataset_id: Identifier for the dataset
            output_dir: Output directory for results
        """
        self._paths = {
            "predictive_matrix": predictive_matrix,
            "stability_file": stability_file,
            "interaction_file": interaction_file,
            "redundancy_file": redundancy_file,
            "removed_file": removed_file
        }
        self._output_dir = Path(output_dir) if output_dir else Path("data/feature_store/selected")
        
        # Create a dummy DataFrame to satisfy BaseResearchModule
        dummy = pd.DataFrame(
            {
                "timestamp": [pd.Timestamp("2000-01-01")]
            }
        )
        
        return super().run_dataframe(
            dummy,
            dataset_id=dataset_id,
            output_dir=self._output_dir
        )
