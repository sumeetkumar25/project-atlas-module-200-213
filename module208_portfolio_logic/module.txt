# atlas/modules/module208_portfolio_logic/module.py
"""
Module 208
Portfolio Logic

Converts signal and confidence into desired portfolio position.

Pipeline: load signal → load confidence → decision engine → position sizing →
          portfolio state machine → statistics → validation → report → export
"""

from typing import Optional, Dict, Any
import pandas as pd
from pathlib import Path

from atlas.core.base_research_module import BaseResearchModule
from atlas.framework.registry import AtlasRegistry

from .engine import PortfolioLogicEngine, PortfolioLogicConfig
from .statistics import PortfolioLogicStatisticsEngine
from .validator import run_validation
from .report import PortfolioLogicReport
from .exporter import PortfolioLogicExporter


class PortfolioLogicResearch(BaseResearchModule):
    """Main research module for portfolio logic."""
    
    MODULE_OUTPUT = "module208"
    METADATA = PortfolioLogicStatisticsEngine.METADATA
    
    def __init__(
        self,
        registry: Optional[AtlasRegistry] = None,
        entry_threshold: float = 0.50,
        exit_threshold: float = 0.20,
        confidence_threshold: float = 0.60,
        max_position_size: float = 1.00,
        allow_position_flip: bool = True,
        timestamp_column: str = "timestamp",
        signal_column: str = "normalized_signal",
        confidence_column: str = "confidence"
    ):
        """
        Initialize the portfolio logic module.
        
        Args:
            registry: Optional Atlas registry
            entry_threshold: Signal threshold to enter a position
            exit_threshold: Signal threshold to exit a position
            confidence_threshold: Minimum confidence to enter a position
            max_position_size: Maximum position size (0-1)
            allow_position_flip: Whether to allow flipping from LONG to SHORT directly
            timestamp_column: Name of the timestamp column
            signal_column: Name of the signal column
            confidence_column: Name of the confidence column
        """
        super().__init__(
            module_name="Portfolio Logic",
            module_version=self.METADATA["version"],
        )
        
        self.registry = registry or AtlasRegistry()
        
        self.config = PortfolioLogicConfig(
            entry_threshold=entry_threshold,
            exit_threshold=exit_threshold,
            confidence_threshold=confidence_threshold,
            max_position_size=max_position_size,
            allow_position_flip=allow_position_flip
        )
        
        self.engine = PortfolioLogicEngine(
            config=self.config,
            timestamp_column=timestamp_column,
            signal_column=signal_column,
            confidence_column=confidence_column
        )
        self.stats_engine = PortfolioLogicStatisticsEngine()
        self.result_data = None
        self.statistics = None
        self.state_history_report = None
        self._output_dir = None
        self._paths = None

    def _register_pipeline_steps(self):
        """Register pipeline steps."""
        self.pipeline_steps = [
            ("portfolio_logic", self._compute_portfolio),
            ("statistics", self._compute_statistics),
            ("validate", self._validate),
            ("export", self._export_results),
        ]
    
    def _prepare_data(self) -> Dict[str, Any]:
        """Prepare data."""
        if self._paths is None:
            raise ValueError("File paths not provided. Use run_portfolio() method.")
        return self._paths
    
    def _compute_portfolio(self) -> Dict[str, Any]:
        """Step 1: Compute portfolio logic."""
        self.logger.info("Step 1: Computing portfolio logic")
        
        prepared = self._prepare_data()
        
        self.result_data = self.engine.compute(
            signal_matrix_path=prepared["signal_matrix"],
            confidence_matrix_path=prepared["confidence_matrix"],
            output_dir=self._output_dir
        )
        
        self.logger.metric(
            "Total Rows",
            f"{self.result_data.summary_statistics['total_rows']:,}"
        )
        self.logger.metric(
            "LONG",
            f"{self.result_data.summary_statistics['long_pct']:.1f}%"
        )
        self.logger.metric(
            "SHORT",
            f"{self.result_data.summary_statistics['short_pct']:.1f}%"
        )
        self.logger.metric(
            "FLAT",
            f"{self.result_data.summary_statistics['flat_pct']:.1f}%"
        )
        self.logger.metric(
            "Position Changes",
            str(self.result_data.summary_statistics['position_changes'])
        )
        self.logger.success("Portfolio logic computed")
        
        # Generate state history report
        self.state_history_report = PortfolioLogicReport.generate_state_history(
            self.result_data.portfolio_state_history,
            top_n=20
        )
        
        return {
            "total_rows": self.result_data.summary_statistics["total_rows"],
            "long_pct": self.result_data.summary_statistics["long_pct"],
            "short_pct": self.result_data.summary_statistics["short_pct"],
            "flat_pct": self.result_data.summary_statistics["flat_pct"],
            "position_changes": self.result_data.summary_statistics["position_changes"]
        }
    
    def _compute_statistics(self) -> Dict[str, Any]:
        """Step 2: Computing statistics."""
        self.logger.info("Step 2: Computing statistics")
        self.statistics = self.stats_engine.analyze(self.result_data)
        
        self.logger.metric(
            "Mean Position Size",
            f"{self.statistics.mean_position_size:.3f}"
        )
        self.logger.metric(
            "Mean Exposure",
            f"{self.statistics.mean_exposure:.3f}"
        )
        self.logger.metric(
            "Avg Holding Time",
            f"{self.statistics.avg_holding_time:.1f} bars"
        )
        self.logger.success("Statistics computed")
        
        return self.stats_engine.summary_dict(self.statistics)
    
    def _validate(self) -> Dict[str, Any]:
        """Step 3: Validation."""
        self.logger.info("Step 3: Validation")
        passed = run_validation(self.result_data, self.statistics)
        self.logger.success("Validation complete")
        return {"software_validation_passed": passed}
    
    def _export_results(self) -> Dict[str, Any]:
        """Step 4: Export."""
        self.logger.info("Step 4: Export")
        
        # Register DataFrames as artifacts
        self.result.artifacts["portfolio_matrix"] = self.result_data.portfolio_matrix
        self.result.artifacts["portfolio_positions"] = self.result_data.portfolio_positions
        self.result.artifacts["portfolio_state_history"] = self.result_data.portfolio_state_history
        
        # Store summary
        stats_df = pd.DataFrame([self.result_data.summary_statistics])
        self.result.artifacts["portfolio_statistics"] = stats_df
        
        # Generate and register reports
        main_report = PortfolioLogicReport.generate(
            self.statistics,
            metadata=self.METADATA,
            dataset_name=self.result.dataset_identifier,
        )
        self.result.artifacts["portfolio_logic_report"] = main_report
        self.result.artifacts["state_history_report"] = self.state_history_report
        
        self.logger.metric("Artifacts", str(len(self.result.artifacts)))
        self.logger.success("Export complete")
        
        return {
            "export_status": "complete",
            "artifact_count": len(self.result.artifacts),
            "total_rows": self.result_data.summary_statistics["total_rows"],
            "position_changes": self.result_data.summary_statistics["position_changes"]
        }
    
    def compute(self) -> Dict[str, Any]:
        """Run the full computation pipeline."""
        self.logger.section("Portfolio Logic Pipeline")
        results = self.execute_pipeline()
        self.result.status = "completed"
        self.logger.success("Module 208 complete")
        return results
    
    def run(self, filepath: str, output_dir=None):
        """Run module on file."""
        self._output_dir = Path(output_dir) if output_dir else None
        return super().run(filepath, output_dir)
    
    def run_dataframe(self, df, dataset_id="in_memory", output_dir=None):
        """Run module on a single DataFrame."""
        self._output_dir = Path(output_dir) if output_dir else None
        return super().run_dataframe(df, dataset_id, output_dir)
    
    def run_portfolio(
        self,
        signal_matrix_path: Path,
        confidence_matrix_path: Path,
        dataset_id: str = "portfolio_logic",
        output_dir: Optional[Path] = None
    ) -> Any:
        """
        Run the portfolio logic.
        
        Args:
            signal_matrix_path: Path to signal_matrix.parquet from Module 206
            confidence_matrix_path: Path to confidence_matrix.parquet from Module 207
            dataset_id: Identifier for the dataset
            output_dir: Output directory for results
        """
        self._paths = {
            "signal_matrix": signal_matrix_path,
            "confidence_matrix": confidence_matrix_path
        }
        self._output_dir = Path(output_dir) if output_dir else Path("data/feature_store/portfolio")
        
        # Create a dummy DataFrame to satisfy BaseResearchModule
        dummy = pd.DataFrame(
            {
                "timestamp": [pd.Timestamp("2000-01-01")]
            }
        )
        
        return super().run_dataframe(
            dummy,
            dataset_id=dataset_id,
            output_dir=self._output_dir
        )
