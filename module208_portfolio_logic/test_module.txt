def create_test_data(n=100, seed=42):
    """Create synthetic test data for portfolio logic."""
    np.random.seed(seed)
    
    timestamps = pd.date_range('2020-01-01', periods=n, freq='1h')
    
    # Signal with known patterns
    signal = np.random.randn(n) * 0.3
    signal[10:30] = 0.7 + np.random.randn(20) * 0.1  # Strong positive
    signal[50:70] = -0.7 + np.random.randn(20) * 0.1  # Strong negative
    signal[80:95] = 0.6 + np.random.randn(15) * 0.1  # Moderate positive
    
    # Confidence
    confidence = np.random.uniform(0.4, 0.9, n)
    confidence[10:30] = 0.85 + np.random.randn(20) * 0.05
    confidence[50:70] = 0.80 + np.random.randn(20) * 0.05
    confidence[80:95] = 0.70 + np.random.randn(15) * 0.05
    
    # Signal matrix
    signal_df = pd.DataFrame({
        'timestamp': timestamps,
        'normalized_signal': signal
    })
    
    # Confidence matrix
    confidence_df = pd.DataFrame({
        'timestamp': timestamps,
        'confidence': np.clip(confidence, 0.4, 1.0)
    })
    
    return {
        'signal_matrix': signal_df,
        'confidence_matrix': confidence_df
    }


def test_engine_load_signal():
    """Test 1: Load signal matrix."""
    print("\n" + "=" * 60)
    print("TEST 1: Load Signal Matrix")
    print("=" * 60)
    
    data = create_test_data()
    engine = PortfolioLogicEngine()
    
    with tempfile.TemporaryDirectory() as tmpdir:
        tmp_path = Path(tmpdir)
        matrix_path = tmp_path / "signal_matrix.parquet"
        data['signal_matrix'].to_parquet(matrix_path, index=False)
        
        loaded_df = engine.load_signal_matrix(matrix_path)
        
        assert not loaded_df.empty
        assert 'timestamp' in loaded_df.columns
        assert 'normalized_signal' in loaded_df.columns
    
    print("  ✓ Load signal matrix test passed")
    return True

def test_engine_desired_position():
    """Test 3: Desired position computation."""
    print("\n" + "=" * 60)
    print("TEST 3: Desired Position")
    print("=" * 60)
    
    config = PortfolioLogicConfig(
        entry_threshold=0.5,
        exit_threshold=0.2,
        confidence_threshold=0.6
    )
    engine = PortfolioLogicEngine(config=config)
    
    # Test scenarios
    signal = pd.Series([0.8, 0.8, 0.3, -0.8, -0.8, -0.3])
    confidence = pd.Series([0.7, 0.5, 0.7, 0.7, 0.5, 0.7])
    
    positions = engine.compute_desired_position(signal, confidence)
    
    # Strong signal with high confidence -> LONG
    assert positions.iloc[0] == 1
    
    # Strong signal with low confidence -> FLAT
    assert positions.iloc[1] == 0
    
    # Weak signal -> FLAT
    assert positions.iloc[2] == 0
    
    # Strong negative with high confidence -> SHORT
    assert positions.iloc[3] == -1
    
    # Strong negative with low confidence -> FLAT
    assert positions.iloc[4] == 0
    
    # Weak negative -> FLAT
    assert positions.iloc[5] == 0
    
    print("  ✓ Desired position test passed")
    return True

def test_engine_state_machine():
    """Test 5: Portfolio state machine."""
    print("\n" + "=" * 60)
    print("TEST 5: Portfolio State Machine")
    print("=" * 60)
    
    config = PortfolioLogicConfig(
        allow_position_flip=True,
        lag_by_one_bar=True
    )
    engine = PortfolioLogicEngine(config=config)
    
    desired_position = pd.Series([1, 1, 0, -1, -1, 1])
    position_size = pd.Series([0.5, 0.5, 0, 0.5, 0.5, 0.5])
    
    state, actual_pos, reason = engine.update_portfolio_state(
        desired_position,
        position_size
    )
    
    # Debug output
    print("\nDesired :", desired_position.tolist())
    print("State   :", state.tolist())
    print("Position:", actual_pos.tolist())
    print("Reason  :", reason.tolist())
    
    # Expected: FLAT -> LONG -> LONG -> FLAT -> SHORT -> SHORT
    expected_states = [
        "FLAT",
        "LONG",
        "LONG",
        "FLAT",
        "SHORT",
        "SHORT",
    ]
    
    for i, exp in enumerate(expected_states):
        assert state.iloc[i] == exp, f"State {i}: expected {exp}, got {state.iloc[i]}"
    
    # Expected positions
    expected_positions = [
        0.0,
        0.5,
        0.5,
        0.0,
        -0.5,
        -0.5,
    ]
    
    for i, exp in enumerate(expected_positions):
        assert actual_pos.iloc[i] == exp, f"Position {i}: expected {exp}, got {actual_pos.iloc[i]}"
    
    # Expected reasons
    expected_reasons = [
        "Initial state (lag enabled)",
        "Entry LONG (lagged)",
        "Hold LONG",
        "Exit LONG (lagged)",
        "Entry SHORT (lagged)",
        "Hold SHORT",
    ]
    
    for i, exp in enumerate(expected_reasons):
        assert reason.iloc[i] == exp, f"Reason {i}: expected '{exp}', got '{reason.iloc[i]}'"
    
    print("  ✓ State machine test passed")
    return True

def test_lag_by_one_bar():
    """Test 12: Lag by one bar comparison."""
    print("\n" + "=" * 60)
    print("TEST 12: Lag by One Bar Comparison")
    print("=" * 60)
    
    # Test with lag disabled (immediate entry)
    config_no_lag = PortfolioLogicConfig(
        allow_position_flip=True,
        lag_by_one_bar=False
    )
    engine_no_lag = PortfolioLogicEngine(config=config_no_lag)
    
    desired_position = pd.Series([1, 1, 0, -1, -1, 1])
    position_size = pd.Series([0.5, 0.5, 0, 0.5, 0.5, 0.5])
    
    state_no_lag, _, _ = engine_no_lag.update_portfolio_state(
        desired_position, position_size
    )
    
    # Test with lag enabled
    config_lag = PortfolioLogicConfig(
        allow_position_flip=True,
        lag_by_one_bar=True
    )
    engine_lag = PortfolioLogicEngine(config=config_lag)
    
    state_lag, _, _ = engine_lag.update_portfolio_state(
        desired_position, position_size
    )
    
    # Debug output
    print("\nNo Lag State  :", state_no_lag.tolist())
    print("Lag State     :", state_lag.tolist())
    
    # First bar should be different
    assert state_no_lag.iloc[0] == "LONG"
    assert state_lag.iloc[0] == "FLAT"
    
    # Lag should shift state sequence by one
    assert state_lag.iloc[1] == state_no_lag.iloc[0]
    
    print("  ✓ Lag by one bar comparison test passed")
    return True
