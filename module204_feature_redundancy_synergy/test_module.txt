def create_test_matrix(n=300, n_features=8, seed=42):
    """Create a synthetic feature matrix for testing."""
    np.random.seed(seed)
    
    timestamps = pd.date_range('2020-01-01', periods=n, freq='1h')
    
    df = pd.DataFrame({'timestamp': timestamps})
    
    # Add close price for target
    close = pd.Series(
        100 + np.cumsum(np.random.randn(n) * 0.5),
        index=df.index
    )
    df['close'] = close
    
    # Create features with known redundancy and synergy
    # Feature 1: Strong predictive
    df['feature1'] = np.random.randn(n) * 10 + 50
    
    # Feature 2: Redundant with feature1 (correlated)
    df['feature2'] = df['feature1'] * 0.9 + np.random.randn(n) * 1
    
    # Feature 3: Synergistic with feature1
    df['feature3'] = np.random.randn(n) * 5 + 50
    
    # Feature 4: Independent
    df['feature4'] = np.random.randn(n) * 8 + 50
    
    # Feature 5: Redundant with feature4
    df['feature5'] = df['feature4'] * 0.85 + np.random.randn(n) * 1
    
    # Feature 6: Synergistic with feature4
    df['feature6'] = np.random.randn(n) * 5 + 50
    
    # Feature 7: Noise
    df['feature7'] = np.random.randn(n) * 10
    
    # Feature 8: Another independent
    df['feature8'] = np.random.randn(n) * 8 + 50
    
    # Target with interactions
    target = (
        0.2 * df['feature1'] + 
        0.1 * df['feature3'] * df['feature1'] / 100 +
        0.2 * df['feature4'] +
        0.1 * df['feature6'] * df['feature4'] / 100 +
        np.random.randn(n) * 2
    )
    df['future_breakout'] = (target > target.mean()).astype(int)
    
    return df


def test_engine_load():
    """Test 1: Load predictive matrix."""
    print("\n" + "=" * 60)
    print("TEST 1: Load Predictive Matrix")
    print("=" * 60)
    
    df = create_test_matrix()
    engine = FeatureRedundancySynergyEngine()
    
    with tempfile.TemporaryDirectory() as tmpdir:
        tmp_path = Path(tmpdir)
        matrix_path = tmp_path / "predictive_feature_matrix.parquet"
        df.to_parquet(matrix_path, index=False)
        
        loaded_df = engine.load_predictive_matrix(matrix_path)
        
        assert not loaded_df.empty
        assert 'timestamp' in loaded_df.columns
        assert len(loaded_df) == len(df)
    
    print("  ✓ Load test passed")
    return True

def test_engine_clustering():
    """Test 6: Feature clustering."""
    print("\n" + "=" * 60)
    print("TEST 6: Feature Clustering")
    print("=" * 60)
    
    df = create_test_matrix()
    engine = FeatureRedundancySynergyEngine()
    
    features = engine.get_feature_columns(df)
    n = len(features)
    redundancy_matrix = np.eye(n)
    
    # Create synthetic redundancy
    for i in range(n):
        for j in range(i+1, n):
            if i < 2 and j < 2:
                redundancy_matrix[i, j] = 0.8
                redundancy_matrix[j, i] = 0.8
            elif i > 3 and j > 3 and i < 6 and j < 6:
                redundancy_matrix[i, j] = 0.7
                redundancy_matrix[j, i] = 0.7
            else:
                redundancy_matrix[i, j] = 0.1
                redundancy_matrix[j, i] = 0.1
    
    redundancy_df = pd.DataFrame(redundancy_matrix, index=features, columns=features)
    clusters = engine.cluster_features(df, features, redundancy_df)
    
    assert not clusters.empty
    assert 'feature' in clusters.columns
    assert 'cluster' in clusters.columns
    
    print("  ✓ Clustering test passed")
    return True
