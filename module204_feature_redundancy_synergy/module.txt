# atlas/modules/module204_feature_redundancy_synergy/module.py
"""
Module 204
Feature Redundancy & Synergy

Evaluates feature groups for redundancy and synergy.

Pipeline: load predictive matrix → compute pair metrics → cluster features →
          make decisions → save optimal matrix
"""

from typing import Optional, Dict, Any
import pandas as pd
from pathlib import Path

from atlas.core.base_research_module import BaseResearchModule
from atlas.framework.registry import AtlasRegistry

from .engine import FeatureRedundancySynergyEngine
from .statistics import FeatureRedundancySynergyStatisticsEngine
from .validator import run_validation
from .report import FeatureRedundancySynergyReport
from .exporter import FeatureRedundancySynergyExporter


class FeatureRedundancySynergyResearch(BaseResearchModule):
    """Main research module for feature redundancy and synergy."""
    
    MODULE_OUTPUT = "module204"
    METADATA = FeatureRedundancySynergyStatisticsEngine.METADATA
    
    def __init__(
        self,
        registry: Optional[AtlasRegistry] = None,
        pass_threshold: float = 0.70,
        warning_threshold: float = 0.50,
        cluster_threshold: float = 0.40,
        random_state: int = 42,
        price_column: str = "close",
        timestamp_column: str = "timestamp",
        target_column: str = "future_breakout",
        n_permutations: int = 10
    ):
        """
        Initialize the feature redundancy and synergy module.
        
        Args:
            registry: Optional Atlas registry
            pass_threshold: Score above this threshold = KEEP
            warning_threshold: Score above this threshold = WARNING
            cluster_threshold: Distance threshold for clustering (0-1)
            random_state: Random seed for reproducibility
            price_column: Name of the price column
            timestamp_column: Name of the timestamp column
            target_column: Name of the target column
            n_permutations: Number of permutations for permutation importance
        """
        super().__init__(
            module_name="Feature Redundancy & Synergy",
            module_version=self.METADATA["version"],
        )
        
        self.registry = registry or AtlasRegistry()
        self.engine = FeatureRedundancySynergyEngine(
            pass_threshold=pass_threshold,
            warning_threshold=warning_threshold,
            cluster_threshold=cluster_threshold,
            random_state=random_state,
            timestamp_column=timestamp_column,
            price_column=price_column,
            target_column=target_column,
            n_permutations=n_permutations
        )
        self.stats_engine = FeatureRedundancySynergyStatisticsEngine()
        self.result_data = None
        self.statistics = None
        self.redundancy_report = None
        self.synergy_report = None
        self._output_dir = None
        self._matrix_path = None

    def _register_pipeline_steps(self):
        """Register pipeline steps."""
        self.pipeline_steps = [
            ("feature_redundancy_synergy", self._compute_redundancy_synergy),
            ("statistics", self._compute_statistics),
            ("validate", self._validate),
            ("export", self._export_results),
        ]
    
    def _prepare_data(self) -> Dict[str, Any]:
        """Prepare data."""
        if self._matrix_path is None:
            raise ValueError("Matrix path not provided. Use run_redundancy_synergy() method.")
        return {"matrix_path": self._matrix_path}
    
    def _compute_redundancy_synergy(self) -> Dict[str, Any]:
        """Step 1: Compute feature redundancy and synergy."""
        self.logger.info("Step 1: Computing feature redundancy and synergy")
        
        prepared = self._prepare_data()
        
        self.result_data = self.engine.compute(
            matrix_path=prepared["matrix_path"],
            output_dir=self._output_dir
        )
        
        self.logger.metric(
            "Total Features",
            str(self.result_data.summary_statistics["total_features"])
        )
        self.logger.metric(
            "KEPT",
            str(self.result_data.summary_statistics["kept_features"])
        )
        self.logger.metric(
            "REMOVED",
            str(self.result_data.summary_statistics["removed_features"])
        )
        self.logger.metric(
            "Clusters",
            str(self.result_data.summary_statistics["unique_clusters"])
        )
        self.logger.metric(
            "Avg Conditional MI",
            f"{self.result_data.summary_statistics['avg_conditional_mi']:.3f}"
        )
        self.logger.success("Feature redundancy and synergy computed")
        
        # Generate reports
        self.redundancy_report = FeatureRedundancySynergyReport.generate_redundancy_table(
            self.result_data.redundancy_table,
            top_n=10
        )
        
        self.synergy_report = FeatureRedundancySynergyReport.generate_synergy_matrix(
            self.result_data.synergy_matrix
        )
        
        return {
            "total_features": self.result_data.summary_statistics["total_features"],
            "kept_features": self.result_data.summary_statistics["kept_features"],
            "removed_features": self.result_data.summary_statistics["removed_features"],
            "unique_clusters": self.result_data.summary_statistics["unique_clusters"],
            "avg_conditional_mi": self.result_data.summary_statistics["avg_conditional_mi"]
        }
    
    def _compute_statistics(self) -> Dict[str, Any]:
        """Step 2: Computing statistics."""
        self.logger.info("Step 2: Computing statistics")
        self.statistics = self.stats_engine.analyze(self.result_data)
        
        self.logger.metric(
            "Avg Synergy",
            f"{self.statistics.avg_synergy:.3f}"
        )
        self.logger.metric(
            "Best Pair",
            self.statistics.best_pair
        )
        self.logger.success("Statistics computed")
        
        return self.stats_engine.summary_dict(self.statistics)
    
    def _validate(self) -> Dict[str, Any]:
        """Step 3: Validation."""
        self.logger.info("Step 3: Validation")
        passed = run_validation(self.result_data, self.statistics)
        self.logger.success("Validation complete")
        return {"software_validation_passed": passed}
    
    def _export_results(self) -> Dict[str, Any]:
        """Step 4: Export."""
        self.logger.info("Step 4: Export")
        
        # Register DataFrames as artifacts
        self.result.artifacts["optimal_matrix"] = self.result_data.optimal_matrix
        self.result.artifacts["redundancy_table"] = self.result_data.redundancy_table
        self.result.artifacts["synergy_matrix"] = self.result_data.synergy_matrix
        self.result.artifacts["conditional_info"] = self.result_data.conditional_info
        self.result.artifacts["feature_clusters"] = self.result_data.feature_clusters
        
        # Store removed features
        removed_df = pd.DataFrame({'removed_features': self.result_data.removed_features})
        self.result.artifacts["removed_features"] = removed_df
        
        # Store summary
        summary_df = pd.DataFrame([self.result_data.summary_statistics])
        self.result.artifacts["module204_statistics"] = summary_df
        
        # Generate and register reports
        main_report = FeatureRedundancySynergyReport.generate(
            self.statistics,
            metadata=self.METADATA,
            dataset_name=self.result.dataset_identifier,
        )
        self.result.artifacts["feature_redundancy_synergy_report"] = main_report
        self.result.artifacts["redundancy_report"] = self.redundancy_report
        self.result.artifacts["synergy_report"] = self.synergy_report
        
        self.logger.metric("Artifacts", str(len(self.result.artifacts)))
        self.logger.success("Export complete")
        
        return {
            "export_status": "complete",
            "artifact_count": len(self.result.artifacts),
            "kept_features": self.result_data.summary_statistics["kept_features"],
            "removed_features": self.result_data.summary_statistics["removed_features"]
        }
    
    def compute(self) -> Dict[str, Any]:
        """Run the full computation pipeline."""
        self.logger.section("Feature Redundancy & Synergy Pipeline")
        results = self.execute_pipeline()
        self.result.status = "completed"
        self.logger.success("Module 204 complete")
        return results
    
    def run(self, filepath: str, output_dir=None):
        """Run module on file."""
        self._output_dir = Path(output_dir) if output_dir else None
        return super().run(filepath, output_dir)
    
    def run_dataframe(self, df, dataset_id="in_memory", output_dir=None):
        """Run module on a single DataFrame."""
        self._output_dir = Path(output_dir) if output_dir else None
        return super().run_dataframe(df, dataset_id, output_dir)
    
    def run_redundancy_synergy(
        self,
        matrix_path: Path,
        dataset_id: str = "feature_redundancy_synergy",
        output_dir: Optional[Path] = None
    ) -> Any:
        """
        Run the feature redundancy and synergy analysis.
        
        Args:
            matrix_path: Path to the predictive feature matrix
            dataset_id: Identifier for the dataset
            output_dir: Output directory for results
        """
        self._matrix_path = matrix_path
        self._output_dir = Path(output_dir) if output_dir else Path("data/feature_store/optimal")
        
        # Create a dummy DataFrame to satisfy BaseResearchModule
        dummy = pd.DataFrame(
            {
                "timestamp": [pd.Timestamp("2000-01-01")]
            }
        )
        
        return super().run_dataframe(
            dummy,
            dataset_id=dataset_id,
            output_dir=self._output_dir
        )
