"""
Module 204 - Feature Redundancy & Synergy Exporter
"""

import pandas as pd
from pathlib import Path


class FeatureRedundancySynergyExporter:
    """Exporter for feature redundancy and synergy results."""
    
    @staticmethod
    def export_optimal_matrix(df: pd.DataFrame, filepath: str):
        """Export optimal matrix to Parquet."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df.to_parquet(path, index=False)
    
    @staticmethod
    def export_redundancy_table(df: pd.DataFrame, filepath: str):
        """Export redundancy table to CSV."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df.to_csv(path, index=False)
    
    @staticmethod
    def export_synergy_matrix(df: pd.DataFrame, filepath: str):
        """Export synergy matrix to CSV."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df.to_csv(path)
    
    @staticmethod
    def export_conditional_info(df: pd.DataFrame, filepath: str):
        """Export conditional information to CSV."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df.to_csv(path, index=False)
    
    @staticmethod
    def export_clusters(df: pd.DataFrame, filepath: str):
        """Export clusters to CSV."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df.to_csv(path, index=False)
    
    @staticmethod
    def export_removed_features(features: list, filepath: str):
        """Export removed features to CSV."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df = pd.DataFrame({'removed_features': features})
        df.to_csv(path, index=False)
    
    @staticmethod
    def export_statistics(stats: dict, filepath: str):
        """Export statistics to CSV."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df = pd.DataFrame([stats])
        df.to_csv(path, index=False)
