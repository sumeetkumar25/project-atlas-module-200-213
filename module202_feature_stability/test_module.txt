def create_test_matrix(n=500, n_features=10):
    """Create a synthetic feature matrix for testing."""
    np.random.seed(42)
    
    timestamps = pd.date_range('2020-01-01', periods=n, freq='1h')
    
    df = pd.DataFrame({'timestamp': timestamps})
    
    # Add close price for regime detection
    close = 100 + np.cumsum(np.random.randn(n) * 0.5)
    df['close'] = close
    df['high'] = close + np.abs(np.random.randn(n) * 1.0)
    df['low'] = close - np.abs(np.random.randn(n) * 1.0)
    
    # Stable feature
    df['stable_feature'] = np.random.randn(n) * 10 + 50
    
    # Unstable feature (mean shifts)
    unstable = np.random.randn(n) * 10 + 50
    unstable[n//2:] += 20
    df['unstable_feature'] = unstable
    
    # Drifting feature (trend)
    df['drifting_feature'] = np.random.randn(n) * 5 + np.linspace(0, 10, n)
    
    # Constant feature
    df['constant_feature'] = np.ones(n) * 5
    
    # More features
    for i in range(5, n_features + 1):
        df[f'feature_{i}'] = np.random.randn(n) * 10 + 50
    
    return df


def test_engine_load():
    """Test 1: Load independent matrix."""
    print("\n" + "=" * 60)
    print("TEST 1: Load Independent Matrix")
    print("=" * 60)
    
    df = create_test_matrix()
    engine = FeatureStabilityEngine()
    
    with tempfile.TemporaryDirectory() as tmpdir:
        tmp_path = Path(tmpdir)
        matrix_path = tmp_path / "independent_feature_matrix.parquet"
        df.to_parquet(matrix_path, index=False)
        
        loaded_df = engine.load_independent_matrix(matrix_path)
        
        assert not loaded_df.empty
        assert 'timestamp' in loaded_df.columns
        assert len(loaded_df) == len(df)
    
    print("  ✓ Load test passed")
    return True


def test_engine_rolling_mean():
    """Test 2: Rolling mean stability."""
    print("\n" + "=" * 60)
    print("TEST 2: Rolling Mean Stability")
    print("=" * 60)
    
    engine = FeatureStabilityEngine(rolling_window=50)
    
    # Stable series
    stable = pd.Series(np.random.randn(500) * 10 + 50)
    stable_metrics = engine.compute_rolling_mean_stability(stable)
    
    # Unstable series (mean shifts)
    unstable = pd.Series(np.random.randn(500) * 10 + 50)
    unstable[250:] += 20
    unstable_metrics = engine.compute_rolling_mean_stability(unstable)
    
    assert stable_metrics['mean_stability'] > unstable_metrics['mean_stability']
    assert 0 <= stable_metrics['mean_stability'] <= 1
    
    print("  ✓ Rolling mean stability test passed")
    return True

def test_engine_distribution_drift():
    """Test 4: Distribution drift."""
    print("\n" + "=" * 60)
    print("TEST 4: Distribution Drift")
    print("=" * 60)
    
    engine = FeatureStabilityEngine()
    
    # Same distribution
    same = pd.Series(np.random.randn(500) * 10 + 50)
    same_drift = engine.compute_distribution_drift(same)
    
    # Different distributions
    first_half = np.random.randn(250) * 10 + 50
    second_half = np.random.randn(250) * 20 + 50
    different = pd.Series(np.concatenate([first_half, second_half]))
    different_drift = engine.compute_distribution_drift(different)
    
    assert same_drift > different_drift
    assert 0 <= same_drift <= 1
    
    print("  ✓ Distribution drift test passed")
    return True
