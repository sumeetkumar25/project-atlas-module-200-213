"""
Module 201
Feature Independence

Computes feature independence and selects representative features.

Pipeline: load integrated matrix → compute dependency metrics → cluster →
          select representatives using composite scores → save independent matrix
"""

from typing import Optional, Dict, Any
import pandas as pd
from pathlib import Path

from atlas.core.base_research_module import BaseResearchModule
from atlas.framework.registry import AtlasRegistry

from .engine import FeatureIndependenceEngine
from .statistics import FeatureIndependenceStatisticsEngine
from .validator import run_validation
from .report import FeatureIndependenceReport
from .exporter import FeatureIndependenceExporter


class FeatureIndependenceResearch(BaseResearchModule):
    """Main research module for feature independence."""
    
    MODULE_OUTPUT = "module201"
    METADATA = FeatureIndependenceStatisticsEngine.METADATA
    
    def __init__(
        self,
        registry: Optional[AtlasRegistry] = None,
        dependency_weights: Optional[Dict[str, float]] = None,
        cluster_method: str = "average",
        cluster_threshold: float = 0.40,
        vif_threshold: float = 10.0,
        n_jobs: int = -1,
        random_state: int = 42,
        quality_scores_path: Optional[Path] = None,
        stability_scores_path: Optional[Path] = None,
        entropy_scores_path: Optional[Path] = None,
        predictive_scores_path: Optional[Path] = None
    ):
        """
        Initialize the feature independence module.
        
        Args:
            registry: Optional Atlas registry
            dependency_weights: Dictionary of weights for dependency metrics
            cluster_method: Linkage method ('ward', 'complete', 'average', 'single')
            cluster_threshold: Distance threshold for clustering (0-1)
            vif_threshold: VIF threshold for removing features
            n_jobs: Number of parallel jobs (-1 = all cores)
            random_state: Random seed for reproducibility
            quality_scores_path: Path to quality scores CSV (Module 119)
            stability_scores_path: Path to stability scores CSV (Module 120)
            entropy_scores_path: Path to entropy scores CSV (Module 109)
            predictive_scores_path: Path to predictive scores CSV (Module 122)
        """
        super().__init__(
            module_name="Feature Independence",
            module_version=self.METADATA["version"],
        )
        
        self.registry = registry or AtlasRegistry()
        
        # Load score DataFrames if paths are provided
        quality_scores = self._load_scores(quality_scores_path)
        stability_scores = self._load_scores(stability_scores_path)
        entropy_scores = self._load_scores(entropy_scores_path)
        predictive_scores = self._load_scores(predictive_scores_path)
        
        self.engine = FeatureIndependenceEngine(
            dependency_weights=dependency_weights,
            cluster_method=cluster_method,
            cluster_threshold=cluster_threshold,
            vif_threshold=vif_threshold,
            n_jobs=n_jobs,
            random_state=random_state,
            quality_scores=quality_scores,
            stability_scores=stability_scores,
            entropy_scores=entropy_scores,
            predictive_scores=predictive_scores
        )
        self.stats_engine = FeatureIndependenceStatisticsEngine()
        self.result_data = None
        self.statistics = None
        self.cluster_report = None
        self._output_dir = None
        self._matrix_path = None

    def _load_scores(self, path: Optional[Path]) -> Optional[pd.DataFrame]:
        """Load scores from CSV file if path exists."""
        if path is not None and path.exists():
            df = pd.read_csv(path)
            if 'feature' in df.columns:
                return df.set_index('feature')
        return None

    def _register_pipeline_steps(self):
        """Register pipeline steps."""
        self.pipeline_steps = [
            ("feature_independence", self._compute_independence),
            ("statistics", self._compute_statistics),
            ("validate", self._validate),
            ("export", self._export_results),
        ]
    
    def _prepare_data(self) -> Dict[str, Any]:
        """Prepare data."""
        if self._matrix_path is None:
            raise ValueError("Matrix path not provided. Use run_independence() method.")
        return {"matrix_path": self._matrix_path}
    
    def _compute_independence(self) -> Dict[str, Any]:
        """Step 1: Compute feature independence."""
        self.logger.info("Step 1: Computing feature independence")
        
        prepared = self._prepare_data()
        
        self.result_data = self.engine.compute(
            matrix_path=prepared["matrix_path"],
            output_dir=self._output_dir
        )
        
        self.logger.metric(
            "Total Features",
            str(self.result_data.summary_statistics["total_features"])
        )
        self.logger.metric(
            "Selected",
            str(self.result_data.summary_statistics["selected_features"])
        )
        self.logger.metric(
            "Removed",
            str(self.result_data.summary_statistics["removed_features"])
        )
        self.logger.metric(
            "Clusters",
            str(self.result_data.summary_statistics["total_clusters"])
        )
        self.logger.success("Feature independence computed")
        
        # Generate report
        self.cluster_report = FeatureIndependenceReport.generate_cluster_report(
            self.result_data.feature_clusters
        )
        
        return {
            "total_features": self.result_data.summary_statistics["total_features"],
            "selected_features": self.result_data.summary_statistics["selected_features"],
            "removed_features": self.result_data.summary_statistics["removed_features"],
            "total_clusters": self.result_data.summary_statistics["total_clusters"]
        }
    
    def _compute_statistics(self) -> Dict[str, Any]:
        """Step 2: Computing statistics."""
        self.logger.info("Step 2: Computing statistics")
        self.statistics = self.stats_engine.analyze(self.result_data)
        
        self.logger.metric(
            "Avg VIF",
            f"{self.statistics.average_vif:.2f}"
        )
        self.logger.metric(
            "Max VIF",
            f"{self.statistics.max_vif:.2f}"
        )
        self.logger.success("Statistics computed")
        
        return self.stats_engine.summary_dict(self.statistics)
    
    def _validate(self) -> Dict[str, Any]:
        """Step 3: Validation."""
        self.logger.info("Step 3: Validation")
        passed = run_validation(self.result_data, self.statistics)
        self.logger.success("Validation complete")
        return {"software_validation_passed": passed}
    
    def _export_results(self) -> Dict[str, Any]:
        """Step 4: Export."""
        self.logger.info("Step 4: Export")
        
        # Register DataFrames as artifacts
        self.result.artifacts["independent_matrix"] = self.result_data.independent_matrix
        self.result.artifacts["dependency_matrix"] = self.result_data.dependency_matrix
        self.result.artifacts["feature_clusters"] = self.result_data.feature_clusters
        
        # Store feature lists as DataFrames
        removed_df = pd.DataFrame({'removed_features': self.result_data.removed_features})
        independent_df = pd.DataFrame({'independent_features': self.result_data.independent_features})
        
        self.result.artifacts["removed_features"] = removed_df
        self.result.artifacts["independent_features"] = independent_df
        
        # Store summary
        summary_df = pd.DataFrame([self.result_data.summary_statistics])
        self.result.artifacts["dependency_statistics"] = summary_df
        
        # Generate and register reports
        main_report = FeatureIndependenceReport.generate(
            self.statistics,
            metadata=self.METADATA,
            dataset_name=self.result.dataset_identifier,
        )
        self.result.artifacts["feature_independence_report"] = main_report
        self.result.artifacts["cluster_report"] = self.cluster_report
        
        self.logger.metric("Artifacts", str(len(self.result.artifacts)))
        self.logger.success("Export complete")
        
        return {
            "export_status": "complete",
            "artifact_count": len(self.result.artifacts),
            "selected_features": self.result_data.summary_statistics["selected_features"],
            "removed_features": self.result_data.summary_statistics["removed_features"]
        }
    
    def compute(self) -> Dict[str, Any]:
        """Run the full computation pipeline."""
        self.logger.section("Feature Independence Pipeline")
        results = self.execute_pipeline()
        self.result.status = "completed"
        self.logger.success("Module 201 complete")
        return results
    
    def run(self, filepath: str, output_dir=None):
        """Run module on file."""
        self._output_dir = Path(output_dir) if output_dir else None
        return super().run(filepath, output_dir)
    
    def run_dataframe(self, df, dataset_id="in_memory", output_dir=None):
        """Run module on a single DataFrame."""
        self._output_dir = Path(output_dir) if output_dir else None
        return super().run_dataframe(df, dataset_id, output_dir)
    
    def run_independence(
        self,
        matrix_path: Path,
        dataset_id: str = "feature_independence",
        output_dir: Optional[Path] = None
    ) -> Any:
        """
        Run the feature independence analysis.
        
        Args:
            matrix_path: Path to the integrated feature matrix
            dataset_id: Identifier for the dataset
            output_dir: Output directory for results
        """
        self._matrix_path = matrix_path
        self._output_dir = Path(output_dir) if output_dir else Path("data/feature_store/independent")
        
        # Create a dummy DataFrame to satisfy BaseResearchModule
        dummy = pd.DataFrame(
            {
                "timestamp": [pd.Timestamp("2000-01-01")]
            }
        )
        
        return super().run_dataframe(
            dummy,
            dataset_id=dataset_id,
            output_dir=self._output_dir
        )
