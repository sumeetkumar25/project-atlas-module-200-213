def create_test_matrix(n=200, n_features=20):
    """Create a synthetic feature matrix for testing."""
    np.random.seed(42)
    
    timestamps = pd.date_range('2020-01-01', periods=n, freq='1h')
    
    # Create base features
    df = pd.DataFrame({'timestamp': timestamps})
    
    # Create correlated groups
    # Group 1: Highly correlated
    base1 = np.random.randn(n) * 10 + 50
    df['feature1'] = base1 + np.random.randn(n) * 0.1
    df['feature2'] = base1 * 0.8 + np.random.randn(n) * 0.2
    df['feature3'] = base1 * 1.2 + np.random.randn(n) * 0.1
    
    # Group 2: Moderately correlated
    base2 = np.random.randn(n) * 5 + 100
    df['feature4'] = base2 + np.random.randn(n) * 0.5
    df['feature5'] = base2 * 0.9 + np.random.randn(n) * 0.5
    
    # Group 3: Independent features
    df['feature6'] = np.random.randn(n) * 10 + 50
    df['feature7'] = np.random.randn(n) * 5 + 100
    df['feature8'] = np.random.randn(n) * 20 + 1000
    
    # More features for larger test
    for i in range(9, n_features + 1):
        df[f'feature{i}'] = np.random.randn(n) * 10 + 50
    
    return df


def test_engine_load():
    """Test 1: Load integrated matrix."""
    print("\n" + "=" * 60)
    print("TEST 1: Load Integrated Matrix")
    print("=" * 60)
    
    df = create_test_matrix()
    engine = FeatureIndependenceEngine()
    
    with tempfile.TemporaryDirectory() as tmpdir:
        tmp_path = Path(tmpdir)
        matrix_path = tmp_path / "integrated_feature_matrix.parquet"
        df.to_parquet(matrix_path, index=False)
        
        loaded_df = engine.load_integrated_matrix(matrix_path)
        
        assert not loaded_df.empty
        assert 'timestamp' in loaded_df.columns
        assert len(loaded_df) == len(df)
    
    print("  ✓ Load test passed")
    return True


def test_engine_pearson():
    """Test 2: Pearson correlation."""
    print("\n" + "=" * 60)
    print("TEST 2: Pearson Correlation")
    print("=" * 60)
    
    df = create_test_matrix(n=100, n_features=8)
    engine = FeatureIndependenceEngine()
    
    features = engine.get_feature_columns(df)
    pearson = engine.compute_pearson_matrix(df, features)
    
    assert not pearson.empty
    assert pearson.shape == (len(features), len(features))
    
    # feature1 and feature2 should be highly correlated
    assert pearson.loc['feature1', 'feature2'] > 0.8
    
    print("  ✓ Pearson test passed")
    return True


def test_engine_mutual_information():
    """Test 3: Mutual Information matrix."""
    print("\n" + "=" * 60)
    print("TEST 3: Mutual Information")
    print("=" * 60)
    
    df = create_test_matrix(n=100, n_features=8)
    engine = FeatureIndependenceEngine()
    
    features = engine.get_feature_columns(df)
    mi = engine.compute_mutual_information_matrix(df, features)
    
    assert not mi.empty
    assert mi.shape == (len(features), len(features))
    assert np.all(mi >= 0)
    assert np.all(mi <= 1)
    
    print("  ✓ Mutual Information test passed")
    return True


def test_engine_vif():
    """Test 4: VIF computation."""
    print("\n" + "=" * 60)
    print("TEST 4: VIF")
    print("=" * 60)
    
    df = create_test_matrix(n=200, n_features=8)
    engine = FeatureIndependenceEngine()
    
    features = engine.get_feature_columns(df)
    vif_df = engine.compute_vif(df, features)
    
    assert not vif_df.empty
    assert 'vif' in vif_df.columns
    assert np.all(vif_df['vif'] >= 1)
    
    # Highly correlated features should have high VIF
    vif1 = vif_df[vif_df['feature'] == 'feature1']['vif'].iloc[0]
    assert vif1 > 1
    
    print("  ✓ VIF test passed")
    return True


def test_engine_dependency_score():
    """Test 5: Dependency score matrix."""
    print("\n" + "=" * 60)
    print("TEST 5: Dependency Score")
    print("=" * 60)
    
    df = create_test_matrix(n=100, n_features=8)
    engine = FeatureIndependenceEngine()
    
    features = engine.get_feature_columns(df)
    pearson = engine.compute_pearson_matrix(df, features)
    spearman = engine.compute_spearman_matrix(df, features)
    mi = engine.compute_mutual_information_matrix(df, features)
    dc = engine.compute_distance_correlation_matrix(df, features)
    
    dependency = engine.compute_dependency_score(pearson, spearman, mi, dc, features)
    
    assert not dependency.empty
    assert dependency.shape == (len(features), len(features))
    assert np.all(dependency >= 0)
    assert np.all(dependency <= 1)
    
    # Diagonal should be 1
    assert np.allclose(np.diag(dependency.values), 1)
    
    print("  ✓ Dependency score test passed")
    return True


def test_engine_clustering():
    """Test 6: Feature clustering."""
    print("\n" + "=" * 60)
    print("TEST 6: Feature Clustering")
    print("=" * 60)
    
    df = create_test_matrix(n=100, n_features=8)
    engine = FeatureIndependenceEngine()
    
    features = engine.get_feature_columns(df)
    pearson = engine.compute_pearson_matrix(df, features)
    spearman = engine.compute_spearman_matrix(df, features)

    print(f"  Found {len(cluster_dict)} clusters")
