"""
Module 212
Monte Carlo Validation

Validates trading system robustness through Monte Carlo simulation.

Pipeline: load trade log → for each simulation:
          shuffle trades → compute equity curve → compute metrics
          → aggregate statistics → export results
"""

from typing import Optional, Dict, Any
import pandas as pd
from pathlib import Path

from atlas.core.base_research_module import BaseResearchModule
from atlas.framework.registry import AtlasRegistry

from .engine import MonteCarloEngine, MonteCarloConfig
from .statistics import MonteCarloStatisticsEngine
from .validator import run_validation
from .report import MonteCarloReport
from .exporter import MonteCarloExporter


# Project results root - single source of truth
PROJECT_ROOT = Path(__file__).resolve().parents[3]
PROJECT_RESULTS = PROJECT_ROOT / "research_results"


class MonteCarloValidation(BaseResearchModule):
    """Main research module for Monte Carlo validation."""
    
    MODULE_OUTPUT = "module212"
    METADATA = MonteCarloStatisticsEngine.METADATA
    
    def __init__(
        self,
        registry: Optional[AtlasRegistry] = None,
        simulations: int = 1000,
        bootstrap: bool = False,
        random_seed: int = 42,
        initial_capital: float = 10000.0
    ):
        """
        Initialize the Monte Carlo validation module.
        
        Args:
            registry: Optional Atlas registry
            simulations: Number of Monte Carlo simulations
            bootstrap: Whether to use bootstrap (sampling with replacement)
            random_seed: Random seed for reproducibility
            initial_capital: Initial capital for equity calculations
        """
        super().__init__(
            module_name="Monte Carlo Validation",
            module_version=self.METADATA["version"],
        )
        
        self.registry = registry or AtlasRegistry()
        
        self.config = MonteCarloConfig(
            simulations=simulations,
            bootstrap=bootstrap,
            random_seed=random_seed,
            initial_capital=initial_capital
        )
        
        self.engine = MonteCarloEngine(config=self.config)
        self.stats_engine = MonteCarloStatisticsEngine()
        self.result_data = None
        self.statistics = None
        self.simulation_summary = None
        self._output_dir = None
        self._paths = None

    def _register_pipeline_steps(self):
        """Register pipeline steps."""
        self.pipeline_steps = [
            ("monte_carlo_validation", self._run_validation),
            ("statistics", self._compute_statistics),
            ("validate", self._validate),
            ("export", self._export_results),
        ]
    
    def _prepare_data(self) -> Dict[str, Any]:
        """
        Prepare input data.
        
        Automatically locates Module 210 output if run_validation() wasn't called.
        """
        if self._paths is not None:
            return self._paths
        
        # Automatically locate Module 210 output
        trade_log = (
            PROJECT_RESULTS
            / "module210_execution_simulation"
            / "executed_trade_log.parquet"
        )
        
        if not trade_log.exists():
            raise FileNotFoundError(
                f"Module210 output not found:\n{trade_log}\n"
                f"Please run Module 210 first."
            )
        
        self._paths = {
            "trade_log": trade_log
        }
        
        self.logger.info(f"Auto-discovered Module210 output: {trade_log}")
        
        return self._paths
    
    def _run_validation(self) -> Dict[str, Any]:
        """Step 1: Run Monte Carlo validation."""
        self.logger.info("Step 1: Running Monte Carlo validation")
        
        prepared = self._prepare_data()
        
        self.result_data = self.engine.compute(
            trade_log_path=prepared["trade_log"],
            output_dir=self._output_dir
        )
        
        self.logger.metric(
            "Simulations",
            str(len(self.result_data.simulation_results))
        )
        self.logger.metric(
            "Probability of Profit",
            f"{self.result_data.summary_statistics.get('prob_profit', 0):.1%}"
        )
        self.logger.metric(
            "Average Sharpe",
            f"{self.result_data.summary_statistics.get('avg_sharpe', 0):.3f}"
        )
        self.logger.metric(
            "Average Return",
            f"{self.result_data.summary_statistics.get('avg_return_pct', 0):.2f}%"
        )
        self.logger.success("Monte Carlo validation complete")
        
        # Generate simulation summary
        self.simulation_summary = MonteCarloReport.generate_simulation_summary(
            self.result_data.simulation_results,
            top_n=10
        )
        
        return {
            "simulations": len(self.result_data.simulation_results),
            "prob_profit": self.result_data.summary_statistics.get("prob_profit", 0),
            "avg_sharpe": self.result_data.summary_statistics.get("avg_sharpe", 0),
            "avg_return": self.result_data.summary_statistics.get("avg_return_pct", 0)
        }
    
    def _compute_statistics(self) -> Dict[str, Any]:
        """Step 2: Computing statistics."""
        self.logger.info("Step 2: Computing statistics")
        self.statistics = self.stats_engine.analyze(self.result_data)
        
        self.logger.metric(
            "Median Equity",
            f"{self.statistics.median_equity:.2f}"
        )
        self.logger.metric(
            "95% VaR",
            f"{self.statistics.p5_equity:.2f}"
        )
        self.logger.metric(
            "Worst Case",
            f"{self.statistics.worst_equity:.2f}"
        )
        self.logger.success("Statistics computed")
        
        return self.stats_engine.summary_dict(self.statistics)
    
    def _validate(self) -> Dict[str, Any]:
        """Step 3: Validation."""
        self.logger.info("Step 3: Validation")
        passed = run_validation(self.result_data, self.statistics)
        self.logger.success("Validation complete")
        return {"software_validation_passed": passed}
    
    def _export_results(self) -> Dict[str, Any]:
        """Step 4: Export."""
        self.logger.info("Step 4: Export")
        
        # Standardized artifacts
        self.result.artifacts["features"] = self.result_data.simulation_results
        self.result.artifacts["statistics"] = pd.DataFrame([self.result_data.summary_statistics])
        self.result.artifacts["report"] = MonteCarloReport.generate(
            self.statistics,
            metadata=self.METADATA,
            dataset_name=self.result.dataset_identifier,
        )
        self.result.artifacts["validation"] = True
        self.result.artifacts["metadata"] = {
            "module": self.MODULE_OUTPUT,
            "version": self.METADATA.get("version", "1.0.0"),
            "dataset": self.result.dataset_identifier,
            "simulations": self.config.simulations,
            "bootstrap": self.config.bootstrap,
            "initial_capital": self.config.initial_capital
        }
        
        # Legacy artifacts
        self.result.artifacts["simulation_results"] = self.result_data.simulation_results
        self.result.artifacts["equity_curves"] = self.result_data.equity_curves
        self.result.artifacts["risk_statistics"] = pd.DataFrame([self.result_data.summary_statistics])
        
        # Generate and register reports
        main_report = MonteCarloReport.generate(
            self.statistics,
            metadata=self.METADATA,
            dataset_name=self.result.dataset_identifier,
        )
        
        # Save report to disk
        report_path = self._output_dir / "monte_carlo_report.txt"
        
        MonteCarloExporter.export_report(
            main_report,
            report_path
        )
        
        self.result.artifacts["monte_carlo_report"] = main_report
        self.result.artifacts["simulation_summary"] = self.simulation_summary
        
        self.logger.metric("Artifacts", str(len(self.result.artifacts)))
        self.logger.success("Export complete")
        
        return {
            "export_status": "complete",
            "artifact_count": len(self.result.artifacts),
            "simulations": len(self.result_data.simulation_results)
        }
    
    def compute(self) -> Dict[str, Any]:
        """Run the full computation pipeline."""
        self.logger.section("Monte Carlo Validation Pipeline")
        results = self.execute_pipeline()
        self.result.status = "completed"
        self.logger.success("Module 212 complete")
        return results
    
    def run(self, filepath: str, output_dir=None):
        """Run module on file."""
        self._output_dir = Path(output_dir) if output_dir else None
        return super().run(filepath, output_dir)
    
    def run_dataframe(self, df, dataset_id="in_memory", output_dir=None):
        """Run module on a single DataFrame."""
        self._output_dir = Path(output_dir) if output_dir else None
        return super().run_dataframe(df, dataset_id, output_dir)
    
    def run_validation(
        self,
        trade_log_path: Path,
        dataset_id: str = "monte_carlo_validation",
        output_dir: Optional[Path] = None
    ) -> Any:
        """
        Run the Monte Carlo validation.
        
        Args:
            trade_log_path: Path to executed_trade_log.parquet from Module 210
            dataset_id: Identifier for the dataset
            output_dir: Output directory for results
        """
        self._paths = {
            "trade_log": trade_log_path
        }
        
        # Use specified output directory or default
        if output_dir is None:
            self._output_dir = PROJECT_RESULTS / "module212_monte_carlo_validation"
        else:
            self._output_dir = Path(output_dir)
        
        # Create a dummy DataFrame to satisfy BaseResearchModule
        dummy = pd.DataFrame(
            {
                "timestamp": [pd.Timestamp("2000-01-01")]
            }
        )
        
        return super().run_dataframe(
            dummy,
            dataset_id=dataset_id,
            output_dir=self._output_dir
        )
