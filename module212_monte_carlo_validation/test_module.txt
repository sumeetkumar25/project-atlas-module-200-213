def create_test_trade_log(n=1000, seed=42):
    """Create synthetic trade log for testing."""
    np.random.seed(seed)
    
    # Generate timestamps
    start_date = datetime(2020, 1, 1)
    timestamps = [start_date + timedelta(hours=i) for i in range(n)]
    
    # Generate trades with varying performance
    directions = np.random.choice(['LONG', 'SHORT'], n, p=[0.6, 0.4])
    entry_prices = 100 + np.random.randn(n) * 10
    exit_prices = entry_prices + np.random.randn(n) * 5
    
    # Add some trend
    trend = np.linspace(0, 20, n)
    exit_prices += trend * 0.1
    
    # Calculate PnL
    pnl = (exit_prices - entry_prices) * 0.5
    pnl *= np.where(directions == 'LONG', 1, -1)
    pnl += np.random.randn(n) * 2
    
    # Adjust prices
    exit_prices = entry_prices + pnl / 0.5
    
    sizes = np.random.uniform(0.1, 1.0, n)
    commissions = np.random.uniform(0.01, 0.05, n)
    slippages = np.random.uniform(0.001, 0.01, n)
    
    trade_log = pd.DataFrame({
        'trade_id': range(n),
        'entry_time': timestamps,
        'exit_time': [ts + timedelta(hours=1) for ts in timestamps],
        'direction': directions,
        'entry_price': entry_prices,
        'exit_price': exit_prices,
        'executed_size': sizes,
        'net_pnl': pnl,
        'total_commission': commissions,
        'total_slippage': slippages,
        'holding_bars': np.random.randint(1, 20, n)
    })
    
    return trade_log


def test_engine_load_trade_log():
    """Test 1: Load trade log."""
    print("\n" + "=" * 60)
    print("TEST 1: Load Trade Log")
    print("=" * 60)
    
    engine = MonteCarloEngine()
    trade_log = create_test_trade_log()
    
    with tempfile.TemporaryDirectory() as tmpdir:
        tmp_path = Path(tmpdir)
        log_path = tmp_path / "trade_log.parquet"
        trade_log.to_parquet(log_path, index=False)
        
        loaded_df = engine.load_trade_log(log_path)
        
        assert not loaded_df.empty
        assert 'entry_time' in loaded_df.columns
        assert 'exit_time' in loaded_df.columns
        assert 'net_pnl' in loaded_df.columns
    
    print("  ✓ Load trade log test passed")
    return True


def test_shuffle_trades():
    """Test 2: Shuffle trades."""
    print("\n" + "=" * 60)
    print("TEST 2: Shuffle Trades")
    print("=" * 60)
    
    config = MonteCarloConfig(bootstrap=False)
    engine = MonteCarloEngine(config=config)
    
    trade_log = create_test_trade_log(n=100)
    shuffled = engine.shuffle_trades(trade_log)
    
    # Check same number of trades
    assert len(shuffled) == len(trade_log)
    
    # Check different order (should be different)
    # Note: There's a tiny chance they could be identical
    if len(trade_log) > 10:
        # Check that at least some order changed
        original_order = trade_log['trade_id'].tolist()
        shuffled_order = shuffled['trade_id'].tolist()
        assert original_order != shuffled_order
    
    print("  ✓ Shuffle trades test passed")
    return True


def test_bootstrap():
    """Test 3: Bootstrap sampling."""
    print("\n" + "=" * 60)
    print("TEST 3: Bootstrap Sampling")
    print("=" * 60)
    
    config = MonteCarloConfig(bootstrap=True)
    engine = MonteCarloEngine(config=config)
    
    trade_log = create_test_trade_log(n=100)
    
    # Run multiple shuffles and check for duplicates
    shuffled1 = engine.shuffle_trades(trade_log)
    shuffled2 = engine.shuffle_trades(trade_log)
    
    # With bootstrap, trades can be repeated
    # Check that we have the same number of trades
    assert len(shuffled1) == len(trade_log)
    assert len(shuffled2) == len(trade_log)
    
    # Check that trade IDs can repeat (with bootstrap)
    trade_ids1 = shuffled1['trade_id'].tolist()
    trade_ids2 = shuffled2['trade_id'].tolist()
    
    # With bootstrap, there should be duplicates
    # Check that we have at least one duplicate in the sample
    has_duplicates1 = len(set(trade_ids1)) < len(trade_ids1)
    has_duplicates2 = len(set(trade_ids2)) < len(trade_ids2)
    
    # It's possible but very unlikely to have no duplicates with n=100 and bootstrap
    print(f"  Bootstrap duplicates present: {has_duplicates1}")
    
    print("  ✓ Bootstrap test passed")
    return True

def test_pipeline():

    print("=" * 60)
    
    trade_log = create_test_trade_log(n=200)
    
    with tempfile.TemporaryDirectory() as tmpdir:
        tmp_path = Path(tmpdir)
        output_dir = tmp_path / "output"

        trade_log.to_parquet(trade_log_path, index=False)
        
        research = MonteCarloValidation(
            simulations=50,
            bootstrap=False,
            random_seed=42,
            initial_capital=10000.0

            dataset_id="pipeline_test",
            output_dir=output_dir
        )
        
        assert result.status == "completed"
        assert "simulation_results" in result.artifacts
        assert "equity_curves" in result.artifacts
        assert "risk_statistics" in result.artifacts
        assert "monte_carlo_report" in result.artifacts
        assert result.statistics.get("software_validation_passed") == True
        
        # Check outputs were saved
        assert (output_dir / "simulation_results.csv").exists()
        assert (output_dir / "equity_curves.parquet").exists()
        assert (output_dir / "risk_statistics.csv").exists()
        assert (output_dir / "monte_carlo_report.txt").exists()
