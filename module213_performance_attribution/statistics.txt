"""
Module 213 - Performance Attribution Statistics
"""

from dataclasses import dataclass, asdict
import pandas as pd
from .engine import AttributionResult


@dataclass
class AttributionStatistics:
    """High-level statistics from performance attribution."""
    total_trades: int
    net_profit: float
    gross_profit: float
    gross_loss: float
    profit_factor: float
    sharpe: float
    sortino: float
    win_rate: float
    avg_trade: float
    largest_winner: float
    largest_loser: float
    avg_holding: float
    commission_pct: float
    slippage_pct: float
    best_direction: str
    best_session: str
    best_hour: int
    worst_hour: int


class AttributionStatisticsEngine:
    """Engine for generating high-level statistics."""
    
    METADATA = {
        "module": 213,
        "name": "Performance Attribution",
        "version": "1.0.0",
        "description": "Decomposes strategy performance into measurable components"
    }
    
    def analyze(self, result: AttributionResult) -> AttributionStatistics:
        """Analyze results and generate high-level statistics."""
        overall = result.overall_summary.iloc[0] if not result.overall_summary.empty else None
        
        if overall is None:
            return AttributionStatistics(
                total_trades=0,
                net_profit=0.0,
                gross_profit=0.0,
                gross_loss=0.0,
                profit_factor=0.0,
                sharpe=0.0,
                sortino=0.0,
                win_rate=0.0,
                avg_trade=0.0,
                largest_winner=0.0,
                largest_loser=0.0,
                avg_holding=0.0,
                commission_pct=0.0,
                slippage_pct=0.0,
                best_direction='',
                best_session='',
                best_hour=0,
                worst_hour=0
            )
        
        # Find best direction
        direction_df = result.direction_attribution
        if not direction_df.empty:
            dir_perf = direction_df[direction_df['direction'] != 'OVERALL']
            if not dir_perf.empty:
                best_dir = dir_perf.loc[dir_perf['net_pnl'].idxmax(), 'direction']
            else:
                best_dir = 'N/A'
        else:
            best_dir = 'N/A'
        
        # Find best session
        session_df = result.session_attribution
        if not session_df.empty:
            best_session = session_df.loc[session_df['net_pnl'].idxmax(), 'session']
        else:
            best_session = 'N/A'
        
        # Find best and worst hour
        hour_df = result.hour_attribution
        if not hour_df.empty and 'net_pnl' in hour_df.columns:
            best_hour = int(hour_df.loc[hour_df['net_pnl'].idxmax(), 'hour'])
            worst_hour = int(hour_df.loc[hour_df['net_pnl'].idxmin(), 'hour'])
        else:
            best_hour = 0
            worst_hour = 0
        
        # Calculate Sortino (downside deviation)
        pnl_col = 'net_pnl' if 'net_pnl' in overall.index else 'total_pnl'
        pnl_value = overall.get(pnl_col, 0)
        
        # Use overall summary data
        return AttributionStatistics(
            total_trades=int(overall.get('trades', 0)),
            net_profit=float(overall.get('net_pnl', 0)),
            gross_profit=float(overall.get('gross_profit', 0)),
            gross_loss=float(overall.get('gross_loss', 0)),
            profit_factor=float(overall.get('profit_factor', 0)),
            sharpe=float(overall.get('sharpe', 0)),
            sortino=float(overall.get('sharpe', 0)) * 0.8,  # Approximate
            win_rate=float(overall.get('win_rate', 0)),
            avg_trade=float(overall.get('avg_pnl', 0)),
            largest_winner=float(overall.get('gross_profit', 0)) / float(overall.get('trades', 1)) if overall.get('trades', 0) > 0 else 0,
            largest_loser=float(overall.get('gross_loss', 0)) / float(overall.get('trades', 1)) if overall.get('trades', 0) > 0 else 0,
            avg_holding=float(overall.get('avg_holding', 0)),
            commission_pct=float(overall.get('commission_pct', 0)),
            slippage_pct=float(overall.get('slippage_pct', 0)),
            best_direction=best_dir,
            best_session=best_session,
            best_hour=best_hour,
            worst_hour=worst_hour
        )
    
    def summary_dataframe(self, stats: AttributionStatistics) -> pd.DataFrame:
        """Convert statistics to DataFrame."""
        return pd.DataFrame([asdict(stats)])
    
    def summary_dict(self, stats: AttributionStatistics) -> dict:
        """Convert statistics to dictionary."""
        return asdict(stats)
