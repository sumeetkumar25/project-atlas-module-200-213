"""
Module 213 - Performance Attribution Validator
"""

import pandas as pd
import numpy as np
from pathlib import Path
from .engine import AttributionResult
from .statistics import AttributionStatistics


class AttributionValidator:
    """Validator for performance attribution results."""

    def validate(
        self,
        result: AttributionResult,
        statistics: AttributionStatistics,
        output_dir: str
    ) -> bool:
        """Validate results and statistics."""
        issues = []
        warnings = []

        # ------------------------------------------------------------
        # Overall summary checks
        # ------------------------------------------------------------
        if result.overall_summary.empty:
            issues.append("Overall summary is empty")

        if not result.overall_summary.empty:
            total_trades = result.overall_summary.iloc[0]['trades']
            if total_trades == 0:
                issues.append("No trades found")

        # ------------------------------------------------------------
        # Full trade-log integrity and financial reconciliation
        # ------------------------------------------------------------
        trade_log = getattr(result, "trade_log", None)

        if trade_log is None or trade_log.empty:
            issues.append("Original trade log is missing.")
        else:
            if trade_log["trade_id"].duplicated().any():
                issues.append("Duplicate trade IDs in original trade log.")

            expected_count = len(trade_log)
            summary_count = int(result.overall_summary.iloc[0]["trades"])

            if expected_count != summary_count:
                issues.append(
                    f"Trade count mismatch: source={expected_count}, "
                    f"summary={summary_count}"
                )

            # Financial reconciliation.
            gross = float(trade_log["gross_pnl_usd"].sum())
            commission = float(trade_log["total_commission"].sum())
            net = float(trade_log["net_pnl_usd"].sum())

            residual = gross - commission - net

            if not np.isclose(residual, 0.0, atol=1e-7, rtol=1e-7):
                issues.append(
                    f"Financial reconciliation failed: residual={residual:.10f}"
                )

            # Direction totals must reconcile with the overall summary.
            directional = result.direction_attribution[
                result.direction_attribution["direction"].isin(
                    ["LONG", "SHORT"]
                )
            ]

            if not directional.empty:
                directional_trades = int(directional["trades"].sum())

                if directional_trades != summary_count:
                    issues.append(
                        f"Directional trade reconciliation failed: "
                        f"LONG+SHORT={directional_trades}, "
                        f"summary={summary_count}"
                    )

        # ------------------------------------------------------------
        # Duplicate trade-ID check across contributor tables
        # ------------------------------------------------------------
        for attr_name in ['top_contributors', 'worst_contributors', 'largest_gains', 'largest_losses']:
            df = getattr(result, attr_name)
            if not df.empty and 'trade_id' in df.columns:
                if df['trade_id'].duplicated().any():
                    issues.append(f"Duplicated trade IDs found in {attr_name}")

        # ------------------------------------------------------------
        # Cost percentage sanity check
        # ------------------------------------------------------------
        if not result.commission_attribution.empty:
            comm_df = result.commission_attribution
            comm_pct = comm_df[comm_df['metric'] == 'Commission %']['value']
            slippage_pct = comm_df[comm_df['metric'] == 'Slippage %']['value']

            if not comm_pct.empty and not slippage_pct.empty:
                total_pct = comm_pct.iloc[0] + slippage_pct.iloc[0]
                if total_pct > 10:
                    warnings.append(f"High total cost: {total_pct:.2f}%")

        # ------------------------------------------------------------
        # NaN check
        # ------------------------------------------------------------
        for attr_name in ['overall_summary', 'direction_attribution', 'hour_attribution', 'day_attribution']:
            df = getattr(result, attr_name)
            if not df.empty:
                if df.isna().any().any():
                    issues.append(f"NaNs found in {attr_name}")

        # ------------------------------------------------------------
        # Exported-file check — upgraded from warning to error
        # ------------------------------------------------------------
        required_files = [
            'performance_summary.csv',
            'performance_by_direction.csv',
            'performance_by_session.csv',
            'performance_by_hour.csv',
            'performance_by_day.csv',
            'performance_by_holding_time.csv',
            'performance_commission.csv',
            'performance_slippage.csv'
        ]

        output_path = Path(output_dir)

        for filename in required_files:
            filepath = output_path / filename
            if not filepath.exists():
                issues.append(f"Required output file missing: {filename}")

        # ------------------------------------------------------------
        # Research-finding warnings (non-fatal)
        # ------------------------------------------------------------
        if statistics.profit_factor < 1.0:
            warnings.append(f"Profit factor below 1.0: {statistics.profit_factor:.3f}")

        if statistics.win_rate < 0.3:
            warnings.append(f"Low win rate: {statistics.win_rate:.1%}")

        if statistics.commission_pct > 2.0:
            warnings.append(f"High commission: {statistics.commission_pct:.2f}%")

        if statistics.slippage_pct > 2.0:
            warnings.append(f"High slippage: {statistics.slippage_pct:.2f}%")

        # ------------------------------------------------------------
        # Report
        # ------------------------------------------------------------
        if warnings:
            print("\n  WARNINGS:")
            for warning in warnings:
                print(f"    ⚠ {warning}")

        if issues:
            raise ValueError(
                f"Performance Attribution failed with {len(issues)} issue(s):\n" +
                "\n".join(f"  - {issue}" for issue in issues)
            )

        return True


def run_validation(result: AttributionResult, statistics: AttributionStatistics, output_dir: str) -> bool:
    """Run validation with user-friendly output."""
    validator = AttributionValidator()

    print("\n" + "=" * 60)
    print("PERFORMANCE ATTRIBUTION VALIDATION")
    print("=" * 60)

    try:
        validator.validate(result, statistics, output_dir)
        print("\n  ✓ All validation checks passed")
        print("=" * 60)
        return True
    except ValueError as e:
        print(e)
        print("=" * 60)
        return False
