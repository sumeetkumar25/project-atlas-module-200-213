"""
Module 213
Performance Attribution

Decomposes strategy performance into measurable components.

Analyses:
- Long vs Short
- Session Attribution
- Holding Time Attribution
- Time-of-Day Attribution
- Day-of-Week Attribution
- Trade Size Attribution
- Volatility Attribution
- Regime Attribution
- Commission Attribution
- Slippage Attribution
"""

from typing import Optional, Dict, Any
from datetime import datetime
import time
import pandas as pd
from pathlib import Path

from atlas.core.base_research_module import BaseResearchModule
from atlas.core.research_result import ResearchResult
from atlas.framework.registry import AtlasRegistry

from .engine import PerformanceAttributionEngine, AttributionConfig
from .statistics import AttributionStatisticsEngine
from .validator import run_validation
from .report import AttributionReport
from .exporter import AttributionExporter


class PerformanceAttribution(BaseResearchModule):
    """Main research module for performance attribution."""
    
    MODULE_OUTPUT = "module213"
    METADATA = AttributionStatisticsEngine.METADATA
    
    def __init__(
        self,
        registry: Optional[AtlasRegistry] = None,
        holding_bins: Optional[list] = None,
        size_bins: Optional[list] = None,
        volatility_bins: Optional[list] = None
    ):
        """
        Initialize the performance attribution module.
        
        Args:
            registry: Optional Atlas registry
            holding_bins: List of holding period bins
            size_bins: List of trade size bins
            volatility_bins: List of volatility bins
        """
        super().__init__(
            module_name="Performance Attribution",
            module_version=self.METADATA["version"],
        )
        
        self.registry = registry or AtlasRegistry()
        
        config = AttributionConfig()
        if holding_bins:
            config.holding_bins = holding_bins
        if size_bins:
            config.size_bins = size_bins
        if volatility_bins:
            config.volatility_bins = volatility_bins
        
        self.engine = PerformanceAttributionEngine(config=config)
        self.stats_engine = AttributionStatisticsEngine()
        self.result_data = None
        self.statistics = None
        self._output_dir = None
        self._paths = None
        self._dataset_id = "performance_attribution"
        self.validation_passed = False

    def _register_pipeline_steps(self):
        """Register pipeline steps."""
        self.pipeline_steps = [
            ("performance_attribution", self._run_attribution),
            ("statistics", self._compute_statistics),
            ("validate", self._validate),
            ("export", self._export_results),
        ]
    
    def _prepare_data(self) -> Dict[str, Any]:
        """Prepare data."""
        if self._paths is None:
            raise ValueError("File paths not provided. Use run_attribution() method.")
        return self._paths
    
    def _run_attribution(self) -> Dict[str, Any]:
        """Step 1: Run performance attribution."""
        self.logger.info("Step 1: Running performance attribution")
        
        prepared = self._prepare_data()
        
        self.result_data = self.engine.compute(
            trade_log_path=prepared["trade_log"],
            output_dir=self._output_dir
        )
        
        overall = self.result_data.overall_summary.iloc[0] if not self.result_data.overall_summary.empty else None
        
        self.logger.metric(
            "Trades",
            str(overall['trades']) if overall is not None else "0"
        )
        self.logger.metric(
            "Net Profit",
            f"{overall['net_pnl']:.2f}" if overall is not None else "0.00"
        )
        self.logger.metric(
            "Profit Factor",
            f"{overall['profit_factor']:.3f}" if overall is not None else "0.000"
        )
        self.logger.metric(
            "Win Rate",
            f"{overall['win_rate']:.1%}" if overall is not None else "0%"
        )
        self.logger.success("Performance attribution complete")
        
        return {
            "trades": overall['trades'] if overall is not None else 0,
            "net_profit": overall['net_pnl'] if overall is not None else 0,
            "profit_factor": overall['profit_factor'] if overall is not None else 0
        }
    
    def _compute_statistics(self) -> Dict[str, Any]:
        """Step 2: Computing statistics."""
        self.logger.info("Step 2: Computing statistics")
        self.statistics = self.stats_engine.analyze(self.result_data)
        
        self.logger.metric(
            "Best Direction",
            self.statistics.best_direction
        )
        self.logger.metric(
            "Commission %",
            f"{self.statistics.commission_pct:.2f}%"
        )
        self.logger.metric(
            "Slippage %",
            f"{self.statistics.slippage_pct:.2f}%"
        )
        self.logger.success("Statistics computed")
        
        return self.stats_engine.summary_dict(self.statistics)
    
    def _validate(self) -> Dict[str, Any]:
        """Step 3: Validate performance attribution."""
        self.logger.info("Step 3: Validation")
        
        passed = run_validation(
            self.result_data,
            self.statistics,
            str(self._output_dir),
        )
        
        self.validation_passed = bool(passed)
        
        if passed:
            self.logger.success("Validation passed")
        else:
            self.logger.error("Validation failed")
        
        return {
            "software_validation_passed": self.validation_passed
        }
    
    def _export_results(self) -> Dict[str, Any]:
        """Step 4: Export."""
        self.logger.info("Step 4: Export")
        
        # Register DataFrames as artifacts
        self.result.artifacts["overall_summary"] = self.result_data.overall_summary
        self.result.artifacts["direction_attribution"] = self.result_data.direction_attribution
        self.result.artifacts["session_attribution"] = self.result_data.session_attribution
        self.result.artifacts["hour_attribution"] = self.result_data.hour_attribution
        self.result.artifacts["day_attribution"] = self.result_data.day_attribution
        self.result.artifacts["holding_time_attribution"] = self.result_data.holding_time_attribution
        self.result.artifacts["size_attribution"] = self.result_data.size_attribution
        self.result.artifacts["volatility_attribution"] = self.result_data.volatility_attribution
        self.result.artifacts["regime_attribution"] = self.result_data.regime_attribution
        self.result.artifacts["commission_attribution"] = self.result_data.commission_attribution
        self.result.artifacts["slippage_attribution"] = self.result_data.slippage_attribution
        self.result.artifacts["top_contributors"] = self.result_data.top_contributors
        self.result.artifacts["worst_contributors"] = self.result_data.worst_contributors
        self.result.artifacts["largest_gains"] = self.result_data.largest_gains
        self.result.artifacts["largest_losses"] = self.result_data.largest_losses
        
        # Generate and register reports
        main_report = AttributionReport.generate(
            self.statistics,
            self.result_data,
            metadata=self.METADATA,
            dataset_name=self.result.dataset_identifier,
            validation_passed=self.validation_passed,
        )
        
        # Save report to disk
        report_path = self._output_dir / "performance_attribution_report.txt"
        AttributionExporter.export_report(main_report, report_path)
        
        self.result.artifacts["performance_attribution_report"] = main_report
        
        self.logger.metric("Artifacts", str(len(self.result.artifacts)))
        self.logger.success("Export complete")
        
        # Report the actual trade count from the statistics engine.
        #
        # The previous version reported len(self.result_data.top_contributors),
        # which is capped at find_top_contributors()'s default n=10 and
        # therefore always reported 10 (or fewer) regardless of how many
        # trades were actually analysed.
        return {
            "export_status": "complete",
            "artifact_count": len(self.result.artifacts),
            "trades": int(self.statistics.total_trades)
        }
    
    def compute(self) -> Dict[str, Any]:
        """Run the full computation pipeline."""
        self.logger.section("Performance Attribution Pipeline")
        results = self.execute_pipeline()
        
        # Validation status must determine the final module status.
        #
        # "completed" means the module ran to completion AND passed its
        # own validation gate. "partial" signals that the pipeline ran
        # but a validation check failed — the runner's mandatory gate
        # will still reject the module, but the status is truthful.
        self.result.status = (
            "completed" if self.validation_passed else "partial"
        )
        
        self.logger.success(
            "Module 213 computation pipeline finished"
        )
        return results
    
    def run(self, filepath: str, output_dir=None):
        """Run module on file."""
        self._output_dir = Path(output_dir) if output_dir else None
        return super().run(filepath, output_dir)
    
    def run_dataframe(self, df, dataset_id="in_memory", output_dir=None):
        """Run module on a single DataFrame."""
        self._output_dir = Path(output_dir) if output_dir else None
        return super().run_dataframe(df, dataset_id, output_dir)
    
    def _execute_attribution_workflow(self):
        """
        Execute Module 213 directly from the Module 210 trade log.

        This deliberately bypasses BaseResearchModule's OHLCV
        validation because Module 213 consumes executed trades,
        not market OHLCV data.
        """
        if self._paths is None or "trade_log" not in self._paths:
            raise ValueError("Module 210 trade log path is missing.")
        if self._output_dir is None:
            raise ValueError("Module 213 output directory is missing.")
        
        self._output_dir.mkdir(parents=True, exist_ok=True)
        
        # Reset market-data fields: this module does not consume OHLCV.
        self.market_df = None
        self.feature_df = None
        self.df = None
        self.result_data = None
        self.statistics = None
        self.validation_passed = False
        
        # Initialize the standard result object without fake market data.
        #
        # timestamp_start / timestamp_end are "N/A" because the module
        # operates on a trade log, not a bar-aligned market DataFrame.
        # The actual trade-date span is exposed via the trade log itself
        # and through the statistics dict.
        self.result = ResearchResult(
            module_name=self.module_name,
            module_version=self.module_version,
            dataset_identifier=str(self._dataset_id),
            timestamp_start="N/A",
            timestamp_end="N/A",
            execution_timestamp=datetime.now().strftime(
                "%Y-%m-%d %H:%M:%S"
            ),
            status="pending",
            statistics={},
            artifacts={},
        )
        
        # Run the pipeline. self.compute() invokes execute_pipeline(),
        # which runs each step in sequence, populating self.result_data,
        # self.statistics, self.validation_passed, and self.result.
        # compute() then sets the final status from validation outcome.
        self.compute()
        
        return self.result
    
    def run_attribution(
        self,
        trade_log_path: Path,
        dataset_id: str = "performance_attribution",
        output_dir: Optional[Path] = None
    ) -> Any:
        """
        Run performance attribution directly from Module 210 trades.
        """
        self._paths = {
            "trade_log": Path(trade_log_path)
        }
        
        self._dataset_id = dataset_id
        
        if not self._paths["trade_log"].exists():
            raise FileNotFoundError(
                f"Module 210 trade log not found: "
                f"{self._paths['trade_log']}"
            )
        
        if output_dir is None:
            self._output_dir = Path(
                "data/feature_store/performance_attribution"
            )
        else:
            self._output_dir = Path(output_dir)
        
        return self._execute_attribution_workflow()
