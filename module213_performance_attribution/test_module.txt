def create_test_trade_log(n=500, seed=42):
    """Create synthetic trade log for testing."""
    np.random.seed(seed)
    
    # Generate timestamps
    start_date = datetime(2020, 1, 1)
    timestamps = [start_date + timedelta(hours=i) for i in range(n)]
    
    # Generate trades with varying performance
    directions = np.random.choice(['LONG', 'SHORT'], n, p=[0.6, 0.4])
    entry_prices = 100 + np.random.randn(n) * 10
    exit_prices = entry_prices + np.random.randn(n) * 5
    
    # Add some trend
    trend = np.linspace(0, 20, n)
    exit_prices += trend * 0.1
    
    # Calculate PnL
    pnl = (exit_prices - entry_prices) * 0.5
    pnl *= np.where(directions == 'LONG', 1, -1)
    pnl += np.random.randn(n) * 2
    
    # Adjust prices
    exit_prices = entry_prices + pnl / 0.5
    
    sizes = np.random.uniform(0.1, 1.0, n)
    commissions = np.random.uniform(0.01, 0.05, n)
    slippages = np.random.uniform(0.001, 0.01, n)
    holding_bars = np.random.randint(1, 30, n)
    
    # Add volatility percentile
    volatility_percentile = np.random.uniform(0, 1, n)
    
    # Add market regime
    regimes = np.random.choice(['Trending', 'Range', 'Expansion', 'Compression'], n)
    
    trade_log = pd.DataFrame({
        'trade_id': range(n),
        'entry_time': timestamps,
        'exit_time': [
            ts + timedelta(hours=int(i))
            for ts, i in zip(timestamps, holding_bars)
        ],
        'direction': directions,
        'entry_price': entry_prices,
        'exit_price': exit_prices,
        'executed_size': sizes,
        'net_pnl': pnl,
        'total_commission': commissions,
        'total_slippage': slippages,
        'holding_bars': holding_bars,
        'volatility_percentile': volatility_percentile,
        'market_regime': regimes
    })
    
    return trade_log


def test_engine_load_trade_log():
    """Test 1: Load trade log."""
    print("\n" + "=" * 60)
    print("TEST 1: Load Trade Log")
    print("=" * 60)
    
    engine = PerformanceAttributionEngine()
    trade_log = create_test_trade_log()
    
    with tempfile.TemporaryDirectory() as tmpdir:
        tmp_path = Path(tmpdir)
        log_path = tmp_path / "trade_log.parquet"
        trade_log.to_parquet(log_path, index=False)
        
        loaded_df = engine.load_trade_log(log_path)
        
        assert not loaded_df.empty
        assert 'entry_time' in loaded_df.columns
        assert 'exit_time' in loaded_df.columns
        assert 'net_pnl' in loaded_df.columns
    
    print("  ✓ Load trade log test passed")
    return True

def test_pipeline():
    """Test 16: Full pipeline."""
    print("\n" + "=" * 60)
    print("TEST 16: Full Pipeline")
    print("=" * 60)
    
    df = create_test_trade_log(n=200)
    
    with tempfile.TemporaryDirectory() as tmpdir:
        tmp_path = Path(tmpdir)
        output_dir = tmp_path / "output"
        
        # Save trade log
        trade_log_path = tmp_path / "trade_log.parquet"
        df.to_parquet(trade_log_path, index=False)
        
        research = PerformanceAttribution()
        
        result = research.run_attribution(
            trade_log_path=trade_log_path,
            dataset_id="pipeline_test",
            output_dir=output_dir
        )
        
        assert result.status == "completed"
        assert "overall_summary" in result.artifacts
        assert "direction_attribution" in result.artifacts
        assert "performance_attribution_report" in result.artifacts
        assert result.statistics.get("software_validation_passed") == True
        
        # Check outputs were saved
        assert (output_dir / "performance_summary.csv").exists()
        assert (output_dir / "performance_by_direction.csv").exists()
        assert (output_dir / "performance_by_session.csv").exists()
        assert (output_dir / "performance_by_hour.csv").exists()
        assert (output_dir / "performance_by_day.csv").exists()
        assert (output_dir / "performance_attribution_report.txt").exists()
    
    print("  ✓ Full pipeline test passed")
    return True
