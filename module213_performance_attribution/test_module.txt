def create_test_trade_log(n=500, seed=42):
    """
    Create synthetic trade log matching Module 210's real trade-log schema.

    The columns here mirror what Module 210 actually exports:
      - executed_size / executed_btc (integer lots, decimal BTC)
      - gross_pnl_usd / net_pnl_usd / net_pnl_inr_est
      - entry_commission / exit_commission / total_commission
      - entry_slippage / exit_slippage / total_slippage

    Note: slippage is embedded in fill prices, so it is NOT subtracted
    from net_pnl here (net = gross - commission only).
    """
    np.random.seed(seed)

    start_date = datetime(2020, 1, 1)
    timestamps = [start_date + timedelta(hours=i) for i in range(n)]

    holding_bars = np.random.randint(1, 30, n)

    # Module 210 emits integer lot counts, not fractional exposure.
    executed_size = np.random.randint(1, 6, n)
    executed_btc = executed_size * 0.001

    entry_prices = np.random.uniform(50000, 100000, n)
    exit_prices = entry_prices + np.random.randn(n) * 500

    directions = np.random.choice(
        ["LONG", "SHORT"], n, p=[0.6, 0.4]
    )

    gross_pnl = (
        (exit_prices - entry_prices)
        * executed_btc
        * np.where(directions == "LONG", 1, -1)
    )
    commissions = (
        (entry_prices + exit_prices)
        * executed_btc
        * 0.0005
    )
    slippages = (
        (entry_prices + exit_prices)
        * executed_btc
        * 0.0001
    )

    # Slippage is already inside gross_pnl because fill prices include
    # adverse slippage. Net = gross - commission.
    net_pnl = gross_pnl - commissions

    # Extra attribution dimensions required by other tests in this suite.
    volatility_percentile = np.random.uniform(0, 1, n)
    regimes = np.random.choice(
        ["Trending", "Range", "Expansion", "Compression"], n
    )

    trade_log = pd.DataFrame({
        "trade_id": range(n),
        "entry_time": timestamps,
        "exit_time": [
            ts + timedelta(hours=int(i))
            for ts, i in zip(timestamps, holding_bars)
        ],
        "direction": directions,
        "entry_price": entry_prices,
        "exit_price": exit_prices,
        "requested_size": executed_size,
        "executed_size": executed_size,
        "executed_btc": executed_btc,
        "entry_commission": commissions / 2,
        "exit_commission": commissions / 2,
        "total_commission": commissions,
        "entry_slippage": slippages / 2,
        "exit_slippage": slippages / 2,
        "total_slippage": slippages,
        "gross_pnl_usd": gross_pnl,
        "net_pnl_usd": net_pnl,
        "net_pnl_inr_est": net_pnl * 90.0,
        "pnl_currency": "USD",
        "pnl_inr_basis_rate": 90.0,
        "holding_bars": holding_bars,
        "exit_reason": ["SIGNAL"] * n,
        # Extra columns consumed by volatility / regime attribution.
        "volatility_percentile": volatility_percentile,
        "market_regime": regimes,
    })

    return trade_log


def test_engine_load_trade_log():
    """Test 1: Load trade log."""
    print("\n" + "=" * 60)
    print("TEST 1: Load Trade Log")
    print("=" * 60)

    engine = PerformanceAttributionEngine()
    trade_log = create_test_trade_log()

    with tempfile.TemporaryDirectory() as tmpdir:
        tmp_path = Path(tmpdir)
        log_path = tmp_path / "trade_log.parquet"
        trade_log.to_parquet(log_path, index=False)

        loaded_df = engine.load_trade_log(log_path)

        assert not loaded_df.empty
        assert 'entry_time' in loaded_df.columns
        assert 'exit_time' in loaded_df.columns
        # The loader normalizes net_pnl_usd -> net_pnl for internal use.
        assert 'net_pnl' in loaded_df.columns
        # The source schema is preserved end-to-end.
        assert 'executed_btc' in loaded_df.columns

    print("  ✓ Load trade log test passed")
    return True

def test_full_engine():
    """Test 13: Full engine with explicit financial reconciliation."""
    print("\n" + "=" * 60)
    print("TEST 13: Full Engine")
    print("=" * 60)

    engine = PerformanceAttributionEngine()
    df = create_test_trade_log(n=200)
    result = engine.compute_attribution(df)

    assert not result.overall_summary.empty
    assert not result.direction_attribution.empty
    assert not result.session_attribution.empty
    assert not result.hour_attribution.empty
    assert not result.day_attribution.empty
    assert not result.holding_time_attribution.empty
    assert not result.size_attribution.empty
    assert not result.volatility_attribution.empty
    assert not result.regime_attribution.empty
    assert not result.commission_attribution.empty
    assert not result.slippage_attribution.empty

    # ------------------------------------------------------------
    # Explicit reconciliation assertions (integration contract).
    # ------------------------------------------------------------
    net_pnl = df["net_pnl_usd"].to_numpy()

    assert np.isclose(
        result.overall_summary.iloc[0]["net_pnl"],
        net_pnl.sum(),
    )

    assert np.isclose(
        result.commission_attribution.loc[
            result.commission_attribution["metric"]
            == "Reconciliation Residual USD",
            "value",
        ].iloc[0],
        0.0,
        atol=1e-7,
    )

    assert "executed_btc" in result.trade_log.columns

    print(f"  Overall trades: {result.overall_summary.iloc[0]['trades']}")
    print("  ✓ Full engine test passed")
    return True

def test_load_trade_log_rejects_corrupt_pnl():
    """
    Test 18: load_trade_log() rejects a log where net_pnl_usd
    does not reconcile with gross_pnl_usd - total_commission.
    """
    print("\n" + "=" * 60)
    print("TEST 18: Load Trade Log Rejects Corrupt PnL")
    print("=" * 60)

    engine = PerformanceAttributionEngine()
    df = create_test_trade_log(n=100)

    # Corrupt exactly one net_pnl_usd value. Reconciliation should fail.
    df.loc[0, "net_pnl_usd"] = df.loc[0, "net_pnl_usd"] + 999.99

    with tempfile.TemporaryDirectory() as tmpdir:
        tmp_path = Path(tmpdir)
        corrupt_path = tmp_path / "corrupt_trade_log.parquet"
        df.to_parquet(corrupt_path, index=False)

        rejected = False
        try:
            engine.load_trade_log(corrupt_path)
        except (ValueError, RuntimeError):
            rejected = True

        assert rejected, (
            "load_trade_log() should have rejected a trade log where "
            "net_pnl_usd does not reconcile with "
            "gross_pnl_usd - total_commission."
        )

    print("  ✓ Corruption rejection test passed")
    return True
