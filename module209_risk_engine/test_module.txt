def create_test_data(n=100, seed=42):
    """Create synthetic test data for risk engine."""
    np.random.seed(seed)
    
    timestamps = pd.date_range('2020-01-01', periods=n, freq='1h')
    
    # Create portfolio data
    signal = np.random.randn(n) * 0.3
    confidence = np.random.uniform(0.4, 0.9, n)
    desired_position = np.random.choice([-1, 0, 1], n, p=[0.2, 0.6, 0.2])
    position_size = np.random.uniform(0.2, 0.8, n)
    actual_position = desired_position * position_size
    
    # Price for volatility
    close = 100 + np.cumsum(np.random.randn(n) * 0.5)
    
    portfolio_df = pd.DataFrame({
        'timestamp': timestamps,
        'signal': signal,
        'confidence': confidence,
        'desired_position': desired_position,
        'position_size': position_size,
        'actual_position': actual_position,
        'exposure': np.abs(actual_position),
        'decision_reason': ['FLAT'] * n,
        'portfolio_state': ['FLAT'] * n,
        'close': close
    })
    
    return {
        'portfolio_matrix': portfolio_df
    }


def test_engine_load_portfolio():
    """Test 1: Load portfolio matrix."""
    print("\n" + "=" * 60)
    print("TEST 1: Load Portfolio Matrix")
    print("=" * 60)
    
    data = create_test_data()
    engine = RiskEngine()
    
    with tempfile.TemporaryDirectory() as tmpdir:
        tmp_path = Path(tmpdir)

        assert not loaded_df.empty
        assert 'timestamp' in loaded_df.columns
        assert 'desired_position' in loaded_df.columns
    
    print("  ✓ Load portfolio matrix test passed")
    return True

def test_risk_reduction_comparison():
    """Test 13: Risk reduction comparison."""
    print("\n" + "=" * 60)
    print("TEST 13: Risk Reduction Comparison")
    print("=" * 60)
    
    data = create_test_data()
    
    # With risk limits
    engine_with_risk = RiskEngine(
        config=RiskConfig(
            max_position_size=0.5,
            max_gross_exposure=0.8,
            leverage_limit=1.0
        )
    )
    
    # Without risk limits
    engine_without_risk = RiskEngine(
        config=RiskConfig(
            max_position_size=1.0,
            max_gross_exposure=1.0,
            leverage_limit=1.0
        )
    )
    
    with tempfile.TemporaryDirectory() as tmpdir:
        tmp_path = Path(tmpdir)
        matrix_path = tmp_path / "portfolio_matrix.parquet"
        data['portfolio_matrix'].to_parquet(matrix_path, index=False)
        
        result_with = engine_with_risk.compute(
            portfolio_matrix_path=matrix_path,
            output_dir=tmp_path / "with_risk"
        )
        
        result_without = engine_without_risk.compute(
            portfolio_matrix_path=matrix_path,
            output_dir=tmp_path / "without_risk"
        )
        
        # With risk should have lower max position
        max_with = result_with.risk_adjusted_portfolio['approved_position'].abs().max()
        max_without = result_without.risk_adjusted_portfolio['approved_position'].abs().max()
        
        assert max_with <= max_without
        
        print(f"  Max position with risk: {max_with:.3f}")
        print(f"  Max position without risk: {max_without:.3f}")
    
    print("  ✓ Risk reduction comparison test passed")
    return True
