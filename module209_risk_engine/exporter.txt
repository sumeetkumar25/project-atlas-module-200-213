"""
Module 209 - Risk Engine Exporter
"""

import pandas as pd
from pathlib import Path


class RiskEngineExporter:
    """Exporter for risk engine results."""
    
    @staticmethod
    def export_risk_adjusted_portfolio(df: pd.DataFrame, filepath: str):
        """Export risk adjusted portfolio to Parquet."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df.to_parquet(path, index=False)
    
    @staticmethod
    def export_approved_positions(df: pd.DataFrame, filepath: str):
        """Export approved positions to CSV."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df.to_csv(path, index=False)
    
    @staticmethod
    def export_risk_events(df: pd.DataFrame, filepath: str):
        """Export risk events to CSV."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df.to_csv(path, index=False)
    
    @staticmethod
    def export_risk_statistics(df: pd.DataFrame, filepath: str):
        """Export risk statistics to CSV."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df.to_csv(path, index=False)
