"""
Module 200 - Feature Integration Statistics
"""

from dataclasses import dataclass, asdict
import pandas as pd
from .engine import FeatureIntegrationResult


@dataclass
class FeatureIntegrationStatistics:
    """High-level statistics from feature integration."""
    total_rows: int
    total_features: int
    removed_duplicates: int
    removed_constants: int
    normalized_features: int
    memory_usage_mb: float
    missing_percentage: float
    feature_density: float
    timestamp_start: str
    timestamp_end: str


class FeatureIntegrationStatisticsEngine:
    """Engine for generating high-level statistics."""
    
    METADATA = {
        "module": 200,
        "name": "Feature Integration",
        "version": "1.0.0",
        "description": "Validates and normalizes master feature matrix"
    }
    
    def analyze(self, result: FeatureIntegrationResult) -> FeatureIntegrationStatistics:
        """Analyze results and generate high-level statistics."""
        return FeatureIntegrationStatistics(
            total_rows=result.summary_statistics['total_rows'],
            total_features=result.summary_statistics['total_features'],
            removed_duplicates=result.summary_statistics['removed_duplicates'],
            removed_constants=result.summary_statistics['removed_constants'],
            normalized_features=result.summary_statistics['normalized_features'],
            memory_usage_mb=result.summary_statistics['memory_usage_mb'],
            missing_percentage=result.summary_statistics['missing_percentage'],
            feature_density=result.summary_statistics['feature_density'],
            timestamp_start=str(result.summary_statistics['timestamp_start']),
            timestamp_end=str(result.summary_statistics['timestamp_end'])
        )
    
    def summary_dataframe(self, stats: FeatureIntegrationStatistics) -> pd.DataFrame:
        """Convert statistics to DataFrame."""
        return pd.DataFrame([asdict(stats)])
    
    def summary_dict(self, stats: FeatureIntegrationStatistics) -> dict:
        """Convert statistics to dictionary."""
        return asdict(stats)
