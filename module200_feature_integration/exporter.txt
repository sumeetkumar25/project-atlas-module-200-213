"""
Module 200 - Feature Integration Exporter
"""

import pandas as pd
from pathlib import Path


class FeatureIntegrationExporter:
    """Exporter for feature integration results."""
    
    @staticmethod
    def export_integrated_matrix(df: pd.DataFrame, filepath: str):
        """Export integrated matrix to Parquet."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df.to_parquet(path, index=False)
    
    @staticmethod
    def export_feature_index(df: pd.DataFrame, filepath: str):
        """Export feature index to CSV."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df.to_csv(path, index=False)
    
    @staticmethod
    def export_summary(summary: dict, filepath: str):
        """Export summary to CSV."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df = pd.DataFrame([summary])
        df.to_csv(path, index=False)
