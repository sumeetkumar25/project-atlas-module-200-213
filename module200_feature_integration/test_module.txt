def create_test_master_matrix():
    """Create a synthetic master feature matrix for testing."""
    n = 100
    np.random.seed(42)
    
    timestamps = pd.date_range('2020-01-01', periods=n, freq='1h')
    
    # Create base features
    df = pd.DataFrame({
        'timestamp': timestamps,
        'feature1': np.random.randn(n) * 10 + 50,
        'feature2': np.random.randn(n) * 5 + 100,
        'feature3': np.random.randn(n) * 2,
        'feature4': np.random.randn(n) * 20 + 1000,
        'feature5': np.ones(n) * 5,  # Constant feature
    })
    
    # Create exact duplicate of feature1
    df['feature6'] = df['feature1']
    
    # Create metadata
    metadata = pd.DataFrame({
        'feature': ['feature1', 'feature2', 'feature3', 'feature4', 'feature5', 'feature6'],
        'dtype': ['float64'] * 6,
        'null_percentage': [0.0] * 6,
        'unique_values': [n] * 5 + [1],
        'min_value': [df['feature1'].min(), df['feature2'].min(), df['feature3'].min(),
                     df['feature4'].min(), 5, df['feature1'].min()],
        'max_value': [df['feature1'].max(), df['feature2'].max(), df['feature3'].max(),
                     df['feature4'].max(), 5, df['feature1'].max()],
        'mean_value': [df['feature1'].mean(), df['feature2'].mean(), df['feature3'].mean(),
                      df['feature4'].mean(), 5, df['feature1'].mean()],
        'std_value': [df['feature1'].std(), df['feature2'].std(), df['feature3'].std(),
                     df['feature4'].std(), 0, df['feature1'].std()]
    })
    
    return df, metadata


def test_engine_load():
    """Test 1: Load master matrix."""
    print("\n" + "=" * 60)
    print("TEST 1: Load Master Matrix")
    print("=" * 60)
    
    df, metadata = create_test_master_matrix()
    engine = FeatureIntegrationEngine()
    
    with tempfile.TemporaryDirectory() as tmpdir:
        tmp_path = Path(tmpdir)
        matrix_path = tmp_path / "master_feature_matrix.parquet"
        metadata_path = tmp_path / "feature_metadata.parquet"
        
        df.to_parquet(matrix_path, index=False)
        metadata.to_parquet(metadata_path, index=False)
        
        loaded_df = engine.load_master_matrix(matrix_path)
        loaded_metadata = engine.load_metadata(metadata_path)
        
        assert not loaded_df.empty
        assert 'timestamp' in loaded_df.columns
        assert len(loaded_df) == len(df)
        assert not loaded_metadata.empty
    
    print("  ✓ Load test passed")
    return True

def test_engine_normalization():

    df, _ = create_test_master_matrix()

def test_engine_feature_index():
    """Test 5: Feature index building."""
    print("\n" + "=" * 60)
    print("TEST 5: Feature Index Building")
    print("=" * 60)
    
    df, metadata = create_test_master_matrix()
    engine = FeatureIntegrationEngine()
    
    df_cleaned, removed_duplicates = engine.remove_duplicate_columns(df)
    df_cleaned, removed_constants = engine.remove_constant_columns(df_cleaned)
    df_normalized, norm_params = engine.normalize_features(df_cleaned)
    
    feature_index = engine.build_feature_index(
        df_normalized, metadata, removed_duplicates, removed_constants, norm_params
    )
    
    assert not feature_index.empty
    assert 'feature' in feature_index.columns
    assert 'normalized' in feature_index.columns
    assert len(feature_index) == len([c for c in df_normalized.columns if c != 'timestamp'])
    
    # Check feature6 is marked as removed_duplicate
    feature6_row = feature_index[feature_index['feature'] == 'feature6']
    if not feature6_row.empty:
        assert feature6_row['removed_duplicate'].iloc[0] == True
    
    print(f"  Index has {len(feature_index)} features")
    print("  ✓ Feature index test passed")
    return True

def test_scalable_duplicate_detection():
    """Test 10: Scalable duplicate detection with many features."""
    print("\n" + "=" * 60)
    print("TEST 10: Scalable Duplicate Detection")
    print("=" * 60)
    
    n = 100
    n_features = 50
    
    np.random.seed(42)
    timestamps = pd.date_range('2020-01-01', periods=n, freq='1h')
    
    # Create DataFrame with many features
    df = pd.DataFrame({'timestamp': timestamps})
    
    # Add features
    for i in range(n_features):
        df[f'feature_{i}'] = np.random.randn(n) * 10 + 50
    
    # Add some duplicates
    df['feature_dup_1'] = df['feature_0']
    df['feature_dup_2'] = df['feature_5']
    df['feature_dup_3'] = df['feature_10']
    
    engine = FeatureIntegrationEngine()
    
    # Count initial features
    initial_features = len([c for c in df.columns if c != 'timestamp'])
    
    # Remove duplicates
    df_cleaned, removed = engine.remove_duplicate_columns(df)
    
    # Verify duplicates were removed
    final_features = len([c for c in df_cleaned.columns if c != 'timestamp'])
    
    assert len(removed) == 3
    assert final_features == initial_features - 3
    
    # Verify the duplicate detection used scalable approach (no O(n²) loops)
    print(f"  Initial features: {initial_features}")
    print(f"  Removed duplicates: {len(removed)}")
    print(f"  Final features: {final_features}")
    print("  ✓ Scalable duplicate detection test passed")
    return True
