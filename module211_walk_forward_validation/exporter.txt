"""
Module 211 - Walk Forward Validation Exporter
"""

import pandas as pd
from pathlib import Path


class WalkForwardExporter:
    """Exporter for walk-forward validation results."""
    
    @staticmethod
    def export_window_results(df: pd.DataFrame, filepath: str):
        """Export window results to CSV."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df.to_csv(path, index=False)
    
    @staticmethod
    def export_equity_curves(df: pd.DataFrame, filepath: str):
        """Export equity curves to Parquet."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df.to_parquet(path, index=False)
    
    @staticmethod
    def export_statistics(df: pd.DataFrame, filepath: str):
        """Export statistics to CSV."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df.to_csv(path, index=False)
    
    @staticmethod
    def export_report(report: str, filepath: str):
        """Export report to text file."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        with open(path, "w", encoding="utf-8") as f:
            f.write(report)
