def create_test_trade_log(n=1000, seed=42):
    """Create synthetic trade log for testing."""
    np.random.seed(seed)
    
    # Generate timestamps
    start_date = datetime(2020, 1, 1)
    timestamps = [start_date + timedelta(hours=i) for i in range(n)]
    
    # Generate trades with varying performance
    directions = np.random.choice(['LONG', 'SHORT'], n, p=[0.6, 0.4])
    entry_prices = 100 + np.random.randn(n) * 10
    exit_prices = entry_prices + np.random.randn(n) * 5
    
    # Add some trend to make it interesting
    trend = np.linspace(0, 20, n)
    exit_prices += trend * 0.1
    
    # Ensure some profitable trades
    pnl = (exit_prices - entry_prices) * 0.5
    pnl *= np.where(directions == 'LONG', 1, -1)
    
    # Randomly assign some profitable and losing trades
    pnl += np.random.randn(n) * 2
    
    # Adjust prices based on PnL
    exit_prices = entry_prices + pnl / 0.5
    
    sizes = np.random.uniform(0.1, 1.0, n)
    commissions = np.random.uniform(0.01, 0.05, n)
    slippages = np.random.uniform(0.001, 0.01, n)
    
    trade_log = pd.DataFrame({
        'trade_id': range(n),
        'entry_time': timestamps,
        'exit_time': [ts + timedelta(hours=1) for ts in timestamps],
        'direction': directions,
        'entry_price': entry_prices,
        'exit_price': exit_prices,
        'executed_size': sizes,
        'total_commission': commissions,
        'total_slippage': slippages,
        'holding_bars': np.random.randint(1, 20, n)
    })
    
    return trade_log


def test_window_generation():
    """Test 1: Window generation."""
    print("\n" + "=" * 60)
    print("TEST 1: Window Generation")
    print("=" * 60)
    
    config = WalkForwardConfig(
        train_window_bars=100,
        test_window_bars=50,
        step_size=50
    )
    engine = WalkForwardEngine(config=config)
    
    data = pd.DataFrame({'dummy': range(500)})
    windows = engine.generate_windows(data)
    
    assert len(windows) > 0
    for train_start, train_end, test_start, test_end in windows:
        assert train_start < train_end
        assert test_start < test_end
        assert train_end <= test_start
        assert test_end <= len(data)
    
    print(f"  Generated {len(windows)} windows")
    print("  ✓ Window generation test passed")
    return True


def test_train_test_split():
    """Test 2: Train/Test split."""
    print("\n" + "=" * 60)
    print("TEST 2: Train/Test Split")
    print("=" * 60)
    
    trade_log = create_test_trade_log(n=500)
    config = WalkForwardConfig(
        train_window_bars=100,
        test_window_bars=50,
        step_size=50,
        minimum_trades=10
    )
    engine = WalkForwardEngine(config=config)
    
    result = engine.run_validation(trade_log)
    
    assert not result.window_results.empty
    
    # Check each window has enough trades
    for _, window in result.window_results.iterrows():
        assert window['n_test_trades'] >= config.minimum_trades
    
    print(f"  Processed {len(result.window_results)} windows")
    print("  ✓ Train/Test split test passed")
    return True


def test_no_overlap():
    """Test 3: No overlap."""
    print("\n" + "=" * 60)
    print("TEST 3: No Overlap")
    print("=" * 60)
    
    config = WalkForwardConfig(
        train_window_bars=100,
        test_window_bars=50,
        step_size=50,
        overlap=False
    )
    engine = WalkForwardEngine(config=config)
    
    trade_log = create_test_trade_log(n=500)
    result = engine.run_validation(trade_log)
    
    # Check no overlap
    for i in range(len(result.window_results) - 1):
        current = result.window_results.iloc[i]
        next_window = result.window_results.iloc[i + 1]
        assert current['test_end'] <= next_window['test_start']
    
    print("  ✓ No overlap test passed")
    return True

def test_walk_forward_consistency():
    """Test 10: Walk-forward consistency."""
    print("\n" + "=" * 60)
    print("TEST 10: Walk-Forward Consistency")
    print("=" * 60)
    
    config = WalkForwardConfig(
        train_window_bars=100,
        test_window_bars=50,
        step_size=50,
        minimum_trades=10
    )
    engine = WalkForwardEngine(config=config)
    
    trade_log = create_test_trade_log(n=1000)
    result = engine.run_validation(trade_log)
    
    # Check consistency across windows
    if len(result.window_results) > 1:
        returns = result.window_results['return_pct']
        
        # Check that returns are not all identical
        assert returns.std() > 0
        
        # Check consistency score
        result.summary_statistics = engine.compute_summary_statistics(result.window_results)
        consistency = result.summary_statistics.get('consistency_score', 0)
        
        print(f"  Consistency Score: {consistency:.3f}")
        print(f"  Positive Windows: {result.summary_statistics.get('positive_window_pct', 0):.1f}%")
    
    print("  ✓ Walk-forward consistency test passed")
    return True
