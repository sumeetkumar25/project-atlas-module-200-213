"""
Module 211 - Walk Forward Validation Validator
"""

import pandas as pd
from .engine import WalkForwardResult
from .statistics import WalkForwardStatistics


class WalkForwardValidator:
    """Validator for walk-forward validation results."""
    
    def __init__(self, timestamp_column: str = "timestamp"):
        self.timestamp_column = timestamp_column
    
    def validate(
        self,
        result: WalkForwardResult,
        statistics: WalkForwardStatistics
    ) -> bool:
        """Validate results and statistics."""
        issues = []
        warnings = []
        
        # Check window results
        if result.window_results.empty:
            warnings.append("No window results available")
        
        # Check equity curves
        if result.equity_curves.empty:
            warnings.append("Equity curves is empty")
        
        # Check for overlapping windows
        if not result.window_results.empty:
            windows = result.window_results.sort_values('window_id')
            
            # Check no overlap
            for i in range(len(windows) - 1):
                current = windows.iloc[i]
                next_window = windows.iloc[i + 1]
                
                # Check test periods don't overlap
                if current['test_end'] > next_window['test_start']:
                    issues.append(
                        f"Overlapping windows: Window {current['window_id']} "
                        f"ends at {current['test_end']} but Window {next_window['window_id']} "
                        f"starts at {next_window['test_start']}"
                    )
                
                # Check no future leakage
                if current['test_start'] < current['train_end']:
                    issues.append(
                        f"Future leakage in Window {current['window_id']}: "
                        f"test starts {current['test_start']} before train ends {current['train_end']}"
                    )
            
            # Check duplicated timestamps
            if 'window_id' in result.window_results.columns:
                if result.window_results['window_id'].duplicated().any():
                    issues.append("Duplicate window IDs found")
            
            # Check for NaNs
            critical_cols = ['return_pct', 'sharpe_ratio', 'max_drawdown', 'win_rate']
            for col in critical_cols:
                if col in result.window_results.columns:
                    if result.window_results[col].isna().any():
                        issues.append(f"NaNs found in {col}")
        
        # Check equity curves
        if not result.equity_curves.empty:
            # Check no missing equity
            if 'equity' in result.equity_curves.columns:
                if result.equity_curves['equity'].isna().any():
                    issues.append("Missing equity values found")
            
            # Check no NaNs in critical columns
            critical_cols = ['equity', 'drawdown', 'window_id']
            for col in critical_cols:
                if col in result.equity_curves.columns:
                    if result.equity_curves[col].isna().any():
                        issues.append(f"NaNs found in equity curve {col}")
        
        # Check statistics
        if statistics.n_windows == 0:
            warnings.append("No windows processed")
        
        if statistics.avg_return == 0 and statistics.n_windows > 0:
            warnings.append("Average return is zero")
        
        if statistics.avg_sharpe < -1:
            warnings.append(f"Very low average Sharpe: {statistics.avg_sharpe:.3f}")
        
        if statistics.positive_window_pct < 30:
            warnings.append(
                f"Low positive window percentage: {statistics.positive_window_pct:.1f}%"
            )
        
        # Warnings (non-fatal)
        if statistics.positive_window_pct < 50:
            warnings.append(
                f"Less than 50% positive windows: {statistics.positive_window_pct:.1f}%"
            )
        
        if statistics.avg_drawdown > 20:
            warnings.append(
                f"High average drawdown: {statistics.avg_drawdown:.1f}%"
            )
        
        if warnings:
            print("\n  WARNINGS:")
            for warning in warnings:
                print(f"    ⚠ {warning}")
        
        if issues:
            raise ValueError(
                f"Walk-Forward Validation failed with {len(issues)} issue(s):\n" +
                "\n".join(f"  - {issue}" for issue in issues)
            )
        
        return True


def run_validation(result: WalkForwardResult, statistics: WalkForwardStatistics) -> bool:
    """Run validation with user-friendly output."""
    validator = WalkForwardValidator()
    
    print("\n" + "=" * 60)
    print("WALK-FORWARD VALIDATION")
    print("=" * 60)
    
    try:
        validator.validate(result, statistics)
        print("\n  ✓ All validation checks passed")
        print("=" * 60)
        return True
    except ValueError as e:
        print(e)
        print("=" * 60)
        return False
